/** Admin authentication and protected admin endpoints. */
const ADMIN_TOKEN_TTL_SECONDS = 1800;

function authenticateAdmin(code) {
  const expected = PropertiesService.getScriptProperties().getProperty('ADMIN_CODE');
  if (!expected) throw new Error('ADMIN_CODE has not been configured in Script Properties.');
  if (String(code || '') !== String(expected)) return {ok: false, message: 'Incorrect administrative code.'};
  const token = Utilities.getUuid();
  CacheService.getScriptCache().put(`ADMIN_TOKEN_${token}`, '1', ADMIN_TOKEN_TTL_SECONDS);
  return {ok: true, token, expiresInMinutes: ADMIN_TOKEN_TTL_SECONDS / 60};
}

function requireAdmin_(token) {
  if (!token || CacheService.getScriptCache().get(`ADMIN_TOKEN_${token}`) !== '1') {
    throw new Error('Administrative session expired. Return to the main page and sign in again.');
  }
  CacheService.getScriptCache().put(`ADMIN_TOKEN_${token}`, '1', ADMIN_TOKEN_TTL_SECONDS);
}

function endAdminSession(token) {
  if (token) CacheService.getScriptCache().remove(`ADMIN_TOKEN_${token}`);
  return true;
}

function runSchemaValidationFromAdmin(token) {
  requireAdmin_(token);
  return validateInventorySchemaV1(false);
}

/** PT and Paper catalog operations. Only this module writes catalog rows. */
const PT_GAME_TYPES = ['Daub', 'Horse Race', 'Under Seal', 'Mixed', 'Other'];
function getPTCatalog_() {
  return getRecords_('PT_Catalog');
}
function buildPTFormOptions_() {
  const options = [];
  getPTCatalog_().filter(game => game.Active === true || String(game.Active).toLowerCase() === 'true').forEach(game => {
    const forms = [normalizeFormNumber_(game.Primary_Form_Number), ...splitAlternateForms_(game.Alternate_Form_Numbers)].filter(Boolean);
    forms.forEach(form => options.push({
      gameId: String(game.PT_Game_ID),
      gameName: String(game.Game_Name),
      formNumber: form,
      label: `${game.Game_Name} (${form})`,
    }));
  });
  return options.sort((a, b) => a.gameName.localeCompare(b.gameName) || a.formNumber.localeCompare(b.formNumber));
}
function assertFormAvailable_(formNumber, excludingGameId) {
  const form = normalizeFormNumber_(formNumber);
  if (!form) throw new Error('Form Number is required.');
  const collision = getPTCatalog_().find(game => {
    if (excludingGameId && String(game.PT_Game_ID) === String(excludingGameId)) return false;
    if (!(game.Active === true || String(game.Active).toLowerCase() === 'true')) return false;
    return [normalizeFormNumber_(game.Primary_Form_Number), ...splitAlternateForms_(game.Alternate_Form_Numbers)].includes(form);
  });
  if (collision) throw new Error(`Form ${form} is already assigned to active game ${collision.Game_Name}.`);
  return form;
}
function createPTGame(token, payload) {
  requireAdmin_(token);

  payload = payload || {};

  const name = normalizeText_(payload.gameName);

  if (!name) {
    throw new Error('Game Name is required.');
  }

  if (!PT_GAME_TYPES.includes(payload.gameType)) {
    throw new Error('Select a valid Game Type.');
  }

  const primary = assertFormAvailable_(payload.primaryFormNumber);
  const alternates = splitAlternateForms_(payload.alternateFormNumbers);

  if (alternates.includes(primary)) {
    throw new Error(
      'Primary Form Number cannot also appear as an alternate.'
    );
  }

  alternates.forEach((form) => {
    assertFormAvailable_(form);
  });

  const cardsPerDeal = integerOrThrow_(
    payload.cardsPerDeal,
    'Cards per Deal',
    { allowZero: false }
  );

  const dealsPerCase = integerOrThrow_(
    payload.dealsPerCase,
    'Deals per Case',
    { allowZero: false }
  );

  const pricePerCard = numberOrThrow_(
    payload.pricePerCard,
    'Price per Card'
  );

  const payoutPerDeal = numberOrThrow_(
    payload.payoutPerDeal,
    'Payout per Deal'
  );

  const manufacturerCostPerDeal = numberOrThrow_(
    payload.manufacturerCostPerDeal,
    'Manufacturer Cost per Deal'
  );

  const grossReceiptsPerDeal = cardsPerDeal * pricePerCard;
  const gamingIncomeBeforeCost = grossReceiptsPerDeal - payoutPerDeal;
  const netIncomePerDeal =
    grossReceiptsPerDeal -
    payoutPerDeal -
    manufacturerCostPerDeal;

  const id = generateId_('PT_GAME', 'PT', 4);

  appendRecord_('PT_Catalog', {
    PT_Game_ID: id,
    Game_Name: name,
    Game_Type: payload.gameType,
    Primary_Form_Number: primary,
    Alternate_Form_Numbers: alternates.join(', '),
    Active: true,
    Display_Order: integerOrThrow_(payload.displayOrder || 0, 'Display Order'),
    Cards_Per_Deal: cardsPerDeal,
    Deals_Per_Case: dealsPerCase,
    Price_Per_Card: pricePerCard,
    Payout_Per_Deal: payoutPerDeal,
    Manufacturer_Cost_Per_Deal: manufacturerCostPerDeal,
    Gross_Receipts_Per_Deal: grossReceiptsPerDeal,
    Gaming_Income_Before_Cost: gamingIncomeBeforeCost,
    Net_Income_Per_Deal: netIncomePerDeal,
    Net_Income_Per_Patron_Dollar: grossReceiptsPerDeal
      ? netIncomePerDeal / grossReceiptsPerDeal
      : 0,
    Manufacturer_Cost_Per_Patron_Dollar: grossReceiptsPerDeal
      ? manufacturerCostPerDeal / grossReceiptsPerDeal
      : 0,
    Manufacturer_Cost_Per_Case: manufacturerCostPerDeal * dealsPerCase,
    Target_On_Hand: integerOrThrow_(payload.targetOnHand || 0, 'Target On Hand'),
    Target_Bundled: integerOrThrow_(payload.targetBundled || 0, 'Target Bundled'),
    Notes: normalizeText_(payload.notes)
  });

  return {
    gameId: id,
    formNumber: primary,
    label: `${name} (${primary})`
  };
}
function addAlternateFormToGame(token, gameId, formNumber) {
  requireAdmin_(token);
  const form = assertFormAvailable_(formNumber, gameId);
  const game = getPTCatalog_().find(row => String(row.PT_Game_ID) === String(gameId));
  if (!game) throw new Error('PT game was not found.');
  const forms = splitAlternateForms_(game.Alternate_Form_Numbers);
  if (normalizeFormNumber_(game.Primary_Form_Number) === form || forms.includes(form)) return {
    gameId, formNumber: form
  };
  forms.push(form);
  updateRecordRow_('PT_Catalog', game._rowNumber, {
    Alternate_Form_Numbers: forms.join(', ')
  });
  return {
    gameId, formNumber: form, label: `${game.Game_Name} (${form})`
  };
}

function createPaperType(token, payload) {
  requireAdmin_(token);

  payload = payload || {};

  const shortName = normalizeText_(payload.shortName).toUpperCase();
  const description = normalizeText_(payload.description);

  if (!shortName || !description) {
    throw new Error(
      'Paper Short Name and Description are required.'
    );
  }

  const existingActivePaper = getPaperCatalog_().find((paper) => {
    return (
      normalizeText_(paper.Short_Name).toUpperCase() === shortName &&
      (
        paper.Active === true ||
        String(paper.Active).toLowerCase() === 'true'
      )
    );
  });

  if (existingActivePaper) {
    throw new Error(
      `Active paper type ${shortName} already exists.`
    );
  }

  const packsPerBox = numberOrThrow_(
    payload.packsPerBox,
    'Packs per Box'
  );

  const boxesPerCase = integerOrThrow_(
    payload.boxesPerCase,
    'Boxes per Case',
    { allowZero: false }
  );

  const packsPerCase = packsPerBox * boxesPerCase;

  if (!Number.isInteger(packsPerCase)) {
    throw new Error(
      'Packs per Box × Boxes per Case must equal a whole number of packs.'
    );
  }

  const manufacturerCostPerCase = numberOrThrow_(
    payload.manufacturerCostPerCase,
    'Manufacturer Cost per Case'
  );

  const salesPricePerPack = numberOrThrow_(
    payload.salesPricePerPack,
    'Sales Price per Pack'
  );

  const manufacturerCostPerPack = packsPerCase
    ? manufacturerCostPerCase / packsPerCase
    : 0;

  const netIncomePerPack = salesPricePerPack - manufacturerCostPerPack;

  const paperId = generateId_(
    'PAPER',
    'PAPER',
    3
  );

  const recordToWrite = {
    Paper_ID: paperId,
    Short_Name: shortName,
    Description: description,
    Active: true,
    Display_Order: integerOrThrow_(payload.displayOrder || 0, 'Display Order'),
    Packs_Per_Box: packsPerBox,
    Boxes_Per_Case: boxesPerCase,
    Packs_Per_Case: packsPerCase,
    Manufacturer_Cost_Per_Case: manufacturerCostPerCase,
    Manufacturer_Cost_Per_Pack: manufacturerCostPerPack,
    Sales_Price_Per_Pack: salesPricePerPack,
    Net_Income_Per_Pack: netIncomePerPack,
    Net_Income_Per_Patron_Dollar: salesPricePerPack
      ? netIncomePerPack / salesPricePerPack
      : 0,
    Manufacturer_Cost_Per_Patron_Dollar: salesPricePerPack
      ? manufacturerCostPerPack / salesPricePerPack
      : 0,
    Target_On_Hand: integerOrThrow_(payload.targetOnHand || 0, 'Target On Hand'),
    Notes: normalizeText_(payload.notes)
  };

  const appendedRowNumber = appendRecord_(
    'Paper_Catalog',
    recordToWrite
  );

  SpreadsheetApp.flush();

  const createdPaper = getPaperCatalog_().find((paper) => {
    return String(paper.Paper_ID) === String(paperId);
  });

  if (!createdPaper) {
    throw new Error(
      `Paper type ${paperId} was written to row ${appendedRowNumber} ` +
      'but could not be read back from Paper_Catalog.'
    );
  }

  return {
    paperId: paperId,
    shortName: shortName,
    label: `${shortName} - ${description}`,
    appendedRowNumber: appendedRowNumber,
    createdPaper: createdPaper
  };
}

/**
 * Returns the spreadsheet to which this Apps Script project is bound.
 *
 * @return {GoogleAppsScript.Spreadsheet.Spreadsheet}
 */
function getInventorySpreadsheet_() {

  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();

  if (!spreadsheet) {

    throw new Error(
      'The bound inventory spreadsheet is unavailable.'
    );
  }

  return spreadsheet;
}


/**
 * Returns a required worksheet.
 *
 * Throws an explicit error rather than allowing later code to fail
 * against a null sheet reference.
 *
 * @param {string} sheetName
 * @return {GoogleAppsScript.Spreadsheet.Sheet}
 */
function getSheetOrThrow_(sheetName) {

  const spreadsheet = getInventorySpreadsheet_();

  const sheet = spreadsheet.getSheetByName(sheetName);

  if (!sheet) {

    throw new Error(
      `Missing required sheet: ${sheetName}`
    );
  }

  return sheet;
}


/**
 * Builds a map between header names and zero-based column indices.
 *
 * Example:
 *
 * {
 *   PT_Game_ID: 0,
 *   Game_Name: 1,
 *   Game_Type: 2
 * }
 *
 * @param {GoogleAppsScript.Spreadsheet.Sheet} sheet
 * @return {Object}
 */
function getHeaderMap_(sheet) {

  const lastColumn = sheet.getLastColumn();

  if (lastColumn < 1) {

    throw new Error(
      `Sheet ${sheet.getName()} has no headers.`
    );
  }

  const headers = sheet
    .getRange(
      1,
      1,
      1,
      lastColumn
    )
    .getDisplayValues()[0];

  return Object.fromEntries(
    headers.map(
      (header, index) => [
        header,
        index
      ]
    )
  );
}


/**
 * Reads all populated records from a worksheet.
 *
 * The returned objects include _rowNumber so that a record can later
 * be updated without depending on hard-coded row positions.
 *
 * @param {string} sheetName
 * @return {Object[]}
 */
function getRecords_(sheetName) {

  const sheet = getSheetOrThrow_(sheetName);

  const lastRow = sheet.getLastRow();

  const lastColumn = sheet.getLastColumn();

  if (
    lastRow < 2 ||
    lastColumn < 1
  ) {

    return [];
  }

  const headers = sheet
    .getRange(
      1,
      1,
      1,
      lastColumn
    )
    .getDisplayValues()[0];

  const values = sheet
    .getRange(
      2,
      1,
      lastRow - 1,
      lastColumn
    )
    .getValues();

  /*
   * Every inventory data table uses column A as its permanent record ID.
   * A formula-only row is not a record when column A is blank.
   */
  return values
    .map(
      (row, rowOffset) => {

        return {
          row: row,
          rowNumber: rowOffset + 2
        };
      }
    )
    .filter(
      (entry) => {

        return normalizeText_(entry.row[0]) !== '';
      }
    )
    .map(
      (entry) => {

        const record = {
          _rowNumber: entry.rowNumber
        };

        headers.forEach(
          (header, index) => {

            record[header] = entry.row[index];
          }
        );

        return record;
      }
    );
}


/**
 * Appends a record to a worksheet using the Google Sheets API.
 *
 * VERSION 1.3 CHANGE:
 *
 * Previous versions used:
 *
 *     sheet.appendRow(row);
 *
 * That can place data below a native Google Sheets Table without
 * extending the Table range. The application could subsequently read
 * the row, but the row would not appear within the visible Table.
 *
 * This implementation uses Sheets.Spreadsheets.Values.append().
 * Google searches the supplied sheet range for its logical table and
 * appends the new row after the existing table data.
 *
 * The Advanced Google Sheets service must remain enabled.
 *
 * @param {string} sheetName
 * @param {Object} record
 * @return {number} Worksheet row number containing the appended record.
 */
function appendRecord_(sheetName, record) {

  const sheet = getSheetOrThrow_(sheetName);

  const lastColumn = sheet.getLastColumn();

  if (lastColumn < 1) {

    throw new Error(
      `Cannot append to ${sheetName}: the sheet has no header columns.`
    );
  }

  const headers = sheet
    .getRange(
      1,
      1,
      1,
      lastColumn
    )
    .getDisplayValues()[0];

  const primaryHeader = headers[0];

  const primaryValue =
    Object.prototype.hasOwnProperty.call(
      record,
      primaryHeader
    )
      ? normalizeText_(record[primaryHeader])
      : '';

  if (!primaryValue) {

    throw new Error(
      `Cannot append to ${sheetName}: ${primaryHeader} is required.`
    );
  }

  const table = getNativeTableForSheet_(sheet);

  if (!table) {

    throw new Error(
      `No native Google Sheets Table was found on ${sheetName}.`
    );
  }

  const targetRow = findFirstEmptyTableRow_(
    sheet,
    table
  );

  const rowValues = headers.map(
    (header) => {

      if (
        Object.prototype.hasOwnProperty.call(
          record,
          header
        )
      ) {

        return record[header];
      }

      return '';
    }
  );

  sheet
    .getRange(
      targetRow,
      1,
      1,
      headers.length
    )
    .setValues([rowValues]);

  SpreadsheetApp.flush();

  return targetRow;
}





/**
 * Returns the native Google Sheets Table defined on a worksheet.
 *
 * Inventory data sheets are designed to contain one native Table each.
 *
 * @param {GoogleAppsScript.Spreadsheet.Sheet} sheet
 * @return {Object|null}
 */
function getNativeTableForSheet_(sheet) {

  const spreadsheet = getInventorySpreadsheet_();

  const apiSpreadsheet = Sheets.Spreadsheets.get(
    spreadsheet.getId(),
    {
      includeGridData: false
    }
  );

  const apiSheet =
    (apiSpreadsheet.sheets || [])
      .find(
        (candidate) => {

          return (
            candidate.properties.sheetId ===
            sheet.getSheetId()
          );
        }
      );

  if (
    !apiSheet ||
    !(apiSheet.tables || []).length
  ) {

    return null;
  }

  return apiSheet.tables[0];
}


/**
 * Finds the first blank permanent-ID row inside a native Table.
 *
 * If the Table contains no empty data row, the Table is expanded by twenty
 * rows and the first newly available row is returned.
 *
 * @param {GoogleAppsScript.Spreadsheet.Sheet} sheet
 * @param {Object} table
 * @return {number}
 */
function findFirstEmptyTableRow_(sheet, table) {

  /*
   * GridRange indices are zero-based and endRowIndex is exclusive.
   * The worksheet's first data row is one row below the Table header.
   */
  const firstDataRow =
    table.range.startRowIndex + 2;

  const lastTableWorksheetRow =
    table.range.endRowIndex;

  const availableRows =
    lastTableWorksheetRow -
    firstDataRow +
    1;

  if (availableRows > 0) {

    const primaryIds = sheet
      .getRange(
        firstDataRow,
        1,
        availableRows,
        1
      )
      .getDisplayValues();

    const emptyOffset = primaryIds.findIndex(
      (row) => {

        return normalizeText_(row[0]) === '';
      }
    );

    if (emptyOffset >= 0) {

      return firstDataRow + emptyOffset;
    }
  }

  const expansionRows = 20;

  const firstNewWorksheetRow =
    lastTableWorksheetRow + 1;

  const expandedLastWorksheetRow =
    lastTableWorksheetRow +
    expansionRows;

  ensureSheetRowCapacity_(
    sheet,
    expandedLastWorksheetRow
  );

  resizeNativeTable_(
    table,
    expandedLastWorksheetRow
  );

  return firstNewWorksheetRow;
}


/**
 * Changes only the ending row of a native Google Sheets Table.
 *
 * @param {Object} table
 * @param {number} newEndRowIndex
 */
function resizeNativeTable_(
  table,
  newEndRowIndex
) {

  const spreadsheet = getInventorySpreadsheet_();

  const updatedTable =
    JSON.parse(
      JSON.stringify(table)
    );

  updatedTable.range.endRowIndex =
    newEndRowIndex;

  Sheets.Spreadsheets.batchUpdate(
    {
      requests: [
        {
          updateTable: {
            table: updatedTable,
            fields: 'range'
          }
        }
      ]
    },
    spreadsheet.getId()
  );
}


/**
 * Adds worksheet rows when a native Table needs additional capacity.
 *
 * @param {GoogleAppsScript.Spreadsheet.Sheet} sheet
 * @param {number} requiredRows
 */
function ensureSheetRowCapacity_(
  sheet,
  requiredRows
) {

  const currentRows = sheet.getMaxRows();

  if (currentRows < requiredRows) {

    sheet.insertRowsAfter(
      currentRows,
      requiredRows - currentRows
    );
  }
}

/**
 * Converts a positive column number into its A1 column letters.
 *
 * Examples:
 *
 *   1  -> A
 *   21 -> U
 *   27 -> AA
 *
 * @param {number} columnNumber
 * @return {string}
 */
function columnNumberToLetter_(columnNumber) {

  let number = Number(columnNumber);

  if (
    !Number.isInteger(number) ||
    number < 1
  ) {

    throw new Error(
      `Invalid column number: ${columnNumber}`
    );
  }

  let columnLetters = '';

  while (number > 0) {

    const remainder =
      (number - 1) % 26;

    columnLetters =
      String.fromCharCode(
        65 + remainder
      ) +
      columnLetters;

    number =
      Math.floor(
        (number - 1) / 26
      );
  }

  return columnLetters;
}


/**
 * Extracts the first worksheet row number from an A1 range.
 *
 * Examples:
 *
 *   PT_Catalog!A21:U21 -> 21
 *   'PT Catalog'!A5:U5 -> 5
 *
 * @param {string} a1Range
 * @return {number|null}
 */
function extractFirstRowNumberFromA1_(a1Range) {

  if (!a1Range) {

    return null;
  }

  const cellRange =
    String(a1Range)
      .split('!')
      .pop();

  const firstCell =
    cellRange
      .split(':')[0];

  const match =
    firstCell.match(
      /(\d+)$/
    );

  if (!match) {

    return null;
  }

  return Number(
    match[1]
  );
}


/**
 * Updates specific fields on an existing record row.
 *
 * @param {string} sheetName
 * @param {number} rowNumber
 * @param {Object} updates
 */
function updateRecordRow_(
  sheetName,
  rowNumber,
  updates
) {

  const sheet =
    getSheetOrThrow_(sheetName);

  const headerMap =
    getHeaderMap_(sheet);


  Object.entries(updates).forEach(
    ([header, value]) => {

      if (!(header in headerMap)) {

        throw new Error(
          `Unknown ${sheetName} column: ${header}`
        );
      }


      sheet
        .getRange(
          rowNumber,
          headerMap[header] + 1
        )
        .setValue(value);
    }
  );
}


/**
 * Generates a persistent unique application ID.
 *
 * Counts are maintained in Script Properties rather than being
 * calculated from worksheet contents. Deleting an old record therefore
 * cannot cause an identifier to be reused.
 *
 * Examples:
 *
 *   generateId_('PT_GAME', 'PT', 4)
 *     -> PT0001
 *
 *   generateId_('TRANSACTION', 'TX-', 7)
 *     -> TX-0000001
 *
 * @param {string} counterName
 * @param {string} prefix
 * @param {number} digits
 * @return {string}
 */
function generateId_(
  counterName,
  prefix,
  digits
) {

  const lock =
    LockService.getScriptLock();

  lock.waitLock(30000);

  try {

    const properties =
      PropertiesService.getScriptProperties();

    const propertyKey =
      `ID_COUNTER_${counterName}`;

    const currentValue =
      Number(
        properties.getProperty(
          propertyKey
        ) || 0
      );

    const nextValue =
      currentValue + 1;


    properties.setProperty(
      propertyKey,
      String(nextValue)
    );


    return (
      prefix +
      String(nextValue).padStart(
        digits,
        '0'
      )
    );

  } finally {

    lock.releaseLock();
  }
}


/**
 * Normalizes general user-entered text.
 *
 * @param {*} value
 * @return {string}
 */
function normalizeText_(value) {

  return String(
    value == null
      ? ''
      : value
  ).trim();
}


/**
 * Normalizes a PT Form Number.
 *
 * Form numbers are stored consistently in uppercase.
 *
 * @param {*} value
 * @return {string}
 */
function normalizeFormNumber_(value) {

  return normalizeText_(
    value
  ).toUpperCase();
}


/**
 * Converts the comma-separated Alternate Form Number field into a
 * normalized, de-duplicated array.
 *
 * @param {*} value
 * @return {string[]}
 */
function splitAlternateForms_(value) {

  const forms =
    normalizeText_(value)
      .split(',')
      .map(
        (formNumber) =>
          normalizeFormNumber_(
            formNumber
          )
      )
      .filter(Boolean);


  return [
    ...new Set(forms)
  ];
}


/**
 * Converts a supplied value into a validated integer.
 *
 * @param {*} value
 * @param {string} label
 * @param {Object=} options
 * @return {number}
 */
function integerOrThrow_(
  value,
  label,
  options
) {

  const number =
    Number(value);

  const allowZero =
    !options ||
    options.allowZero !== false;


  const invalidInteger =
    !Number.isInteger(number);

  const invalidPositiveValue =
    !allowZero &&
    number <= 0;

  const invalidNonnegativeValue =
    allowZero &&
    number < 0;


  if (
    invalidInteger ||
    invalidPositiveValue ||
    invalidNonnegativeValue
  ) {

    throw new Error(
      `${label} must be ` +
      `${allowZero ? 'a nonnegative' : 'a positive'} ` +
      'whole number.'
    );
  }


  return number;
}


/**
 * Converts a supplied value into a validated nonnegative number.
 *
 * @param {*} value
 * @param {string} label
 * @return {number}
 */
function numberOrThrow_(
  value,
  label
) {

  const number =
    Number(value);


  if (
    !Number.isFinite(number) ||
    number < 0
  ) {

    throw new Error(
      `${label} must be a nonnegative number.`
    );
  }


  return number;
}

 * changes easier and keeps business modules consistent.
 *
 * @return {Date}
 */
function now_() {

  return new Date();
}


/**
 * One-time V1.3 recovery helper.
 *
 * The first test PT game was written below the original PT_Catalog
 * native Table because the pre-V1.3 appendRecord_() used appendRow().
 *
 * This helper expands the PT_Catalog native table sufficiently to
 * include all currently populated worksheet rows.
 *
 * Run this ONCE after installing V1.3 if PT_Catalog contains a record
 * below the current native Table range.
 *
 * This helper does not move, modify, or duplicate any catalog data.
 */
function recoverPTCatalogTableRangeV13() {

  const spreadsheet =
    getInventorySpreadsheet_();

  const sheet =
    getSheetOrThrow_(
      'PT_Catalog'
    );

  const lastDataRow =
    sheet.getLastRow();


  if (lastDataRow < 2) {

    return {
      changed: false,
      message: 'PT_Catalog contains no data rows to recover.'
    };
  }


  const apiSpreadsheet =
    Sheets.Spreadsheets.get(

      spreadsheet.getId(),

      {
        includeGridData: false
      }
    );


  const apiSheet =
    (apiSpreadsheet.sheets || [])
      .find(
        (candidate) => {

          return (
            candidate.properties.sheetId ===
            sheet.getSheetId()
          );
        }
      );


  if (!apiSheet) {

    throw new Error(
      'PT_Catalog API sheet metadata could not be found.'
    );
  }


  const table =
    (apiSheet.tables || [])
      .find(
        (candidate) => {

          return (
            candidate.name ===
            'PT_Catalog_Table'
          );
        }
      );


  if (!table) {

    throw new Error(
      'PT_Catalog_Table could not be found.'
    );
  }


  /*
   * API table ranges use an exclusive endRowIndex.
   *
   * Worksheet row 21 therefore requires endRowIndex >= 21.
   */
  const currentEndRowIndex =
    table.range.endRowIndex || 0;


  if (
    currentEndRowIndex >=
    lastDataRow
  ) {

    return {
      changed: false,
      message:
        'PT_Catalog_Table already includes all populated rows.'
    };
  }


  const updatedTable =
    JSON.parse(
      JSON.stringify(table)
    );


  updatedTable.range.endRowIndex =
    lastDataRow;


  Sheets.Spreadsheets.batchUpdate(

    {
      requests: [
        {
          updateTable: {
            table: updatedTable,
            fields: 'range'
          }
        }
      ]
    },

    spreadsheet.getId()
  );


  return {
    changed: true,
    lastDataRow: lastDataRow,
    message:
      `PT_Catalog_Table was expanded through worksheet row ${lastDataRow}.`
  };
}

/**

function findPTDeal_(formNumber, serialNumber) {
  const normalizedFormNumber = normalizeFormNumber_(formNumber);
  const normalizedSerialNumber = normalizeText_(serialNumber);

  if (!normalizedSerialNumber) {
    return null;
  }

  return getPTDeals_().find((deal) => {
    return (
      normalizeFormNumber_(deal.Form_Number) === normalizedFormNumber &&
      normalizeText_(deal.Serial_Number) === normalizedSerialNumber
    );
  }) || null;

/**
 * Creates one physical PT deal record.
 *
 * Serial_Number may be blank while the deal is unidentified. Duplicate
 * protection is enforced only after a nonblank serial number exists.
 */
function createPTDeal_(data) {
  if (!PT_DEAL_STATUSES.includes(data.status)) {
    throw new Error(`Invalid PT Deal Status: ${data.status}`);
  }

  const normalizedFormNumber = normalizeFormNumber_(data.formNumber);
  const normalizedSerialNumber = normalizeText_(data.serialNumber);

  if (
    normalizedSerialNumber &&
    findPTDeal_(normalizedFormNumber, normalizedSerialNumber)
  ) {
    throw new Error(
      `Deal ${normalizedFormNumber}/${normalizedSerialNumber} already exists.`
    );
  }

  const dealId = generateId_(
    'PT_DEAL',
    'PTD-',
    6
  );

  const receivedDate = data.receivedDate || now_();

  appendRecord_('PT_Deals', {
    PT_Deal_ID: dealId,
    PT_Game_ID: data.gameId,
    Form_Number: normalizedFormNumber,
    Serial_Number: normalizedSerialNumber,
    Received_Date: receivedDate,
    Deal_Status: data.status,
    Bundled_Date: data.status === 'Bundled' ? receivedDate : '',
    Expended_Date: [
      'Sold',
      'Destroyed',
      'Returned'
    ].includes(data.status) ? receivedDate : '',
    Expended_Session_ID: data.sessionId || '',
    Notes: data.notes || ''
  });

  return dealId;
}


function updatePTDealStatus_(deal, newStatus) {
  if (!PT_DEAL_STATUSES.includes(newStatus)) {
    throw new Error(`Invalid PT Deal Status: ${newStatus}`);
  }

  const updates = {
    Deal_Status: newStatus
  };

  if (
    newStatus === 'Bundled' &&
    !deal.Bundled_Date
  ) {
    updates.Bundled_Date = now_();
  }

  if (
    [
      'Sold',
      'Destroyed',
      'Returned'
    ].includes(newStatus) &&
    !deal.Expended_Date
  ) {
    updates.Expended_Date = now_();
  }

  updateRecordRow_(
    'PT_Deals',
    deal._rowNumber,
    updates
  );
}

/**
 * Pre-validates a PT stock addition before any records are written.
 *
 * This is important for a multi-deal order. All product-level conditions are
 * checked once before the first deal or transaction is created.
 */
function validateAnonymousPTStockAddition_(data) {
  const quantity = integerOrThrow_(
    data.quantity,
    'PT quantity',
    { allowZero: false }
  );

  const gameId = normalizeText_(data.gameId);
  const formNumber = normalizeFormNumber_(data.formNumber);

  if (!gameId) {
    throw new Error('PT Game ID is required.');
  }

  if (!formNumber) {
    throw new Error('PT Form Number is required.');
  }

  const game = getPTCatalog_().find((catalogRow) => {
    return String(catalogRow.PT_Game_ID) === gameId;
  });

  if (!game) {
    throw new Error(
      `PT Game ${gameId} was not found in PT_Catalog.`
    );
  }

  const validForms = [
    normalizeFormNumber_(game.Primary_Form_Number),
    ...splitAlternateForms_(game.Alternate_Form_Numbers)
  ];

  if (!validForms.includes(formNumber)) {
    throw new Error(
      `Form ${formNumber} is not assigned to ${game.Game_Name}.`
    );
  }

  if (
    data.status &&
    data.status !== 'Unbundled'
  ) {
    throw new Error(
      'Anonymous ordered PT stock must enter inventory as Unbundled.'
    );
  }

  return {
    quantity: quantity,
    gameId: gameId,
    formNumber: formNumber
  };
}


/**
 * Adds multiple anonymous PT deals.
 *
 * Validation is completed before the first write occurs. Blank serial numbers
 * are legitimate and can occur on many deals of the same Form Number.
 */
function createAnonymousPTStock_(data) {
  const validated = validateAnonymousPTStockAddition_(data);
  const createdRecords = [];

  try {
    for (
      let dealNumber = 0;
      dealNumber < validated.quantity;
      dealNumber += 1
    ) {
      const dealId = createPTDeal_({
        gameId: validated.gameId,
        formNumber: validated.formNumber,
        serialNumber: '',
        status: 'Unbundled',
        receivedDate: data.effectiveDate || now_(),
        notes: data.notes || ''
      });

      createdRecords.push({
        dealId: dealId,
        transactionId: ''
      });

      const transactionId = postInventoryTransaction_({
        effectiveDate: data.effectiveDate || now_(),
        inventoryType: 'PT',
        productId: validated.gameId,
        itemId: dealId,
        formNumber: validated.formNumber,
        serialNumber: '',
        transactionType: data.transactionType || 'Received',
        quantityChange: 1,
        unit: 'Deal',
        source: data.source,
        reconciliationId: data.reconciliationId || '',
        notes: data.notes || ''
      });

      createdRecords[createdRecords.length - 1].transactionId = transactionId;
    }
  } catch (error) {
    rollbackNewPTStockRecords_(createdRecords);

    throw new Error(
      `PT stock addition failed and was rolled back: ${error.message}`
    );
  }

  return createdRecords.map((record) => record.dealId);
}


/**
 * Best-effort rollback used only for records created during the current failed
 * multi-deal operation.
 *
 * This does not rewrite historical inventory. It exists to prevent an order
 * from being partially posted if an unexpected application/API failure occurs
 * after pre-validation succeeds.
 */
function rollbackNewPTStockRecords_(createdRecords) {
  if (!createdRecords.length) {
    return;
  }

  const transactionIds = new Set(
    createdRecords
      .map((record) => record.transactionId)
      .filter(Boolean)
  );

  const dealIds = new Set(
    createdRecords
      .map((record) => record.dealId)
      .filter(Boolean)
  );

  deleteRowsByIdSet_(
    'Transactions',
    'Transaction_ID',
    transactionIds
  );

  deleteRowsByIdSet_(
    'PT_Deals',
    'PT_Deal_ID',
    dealIds
  );
}


/**
 * Deletes rows matching IDs created by the currently failing operation.
 * Rows are deleted from bottom to top to avoid row-number shifting.
 */
function deleteRowsByIdSet_(
  sheetName,
  idColumnName,
  ids
) {
  if (!ids || ids.size === 0) {
    return;
  }

  const sheet = getSheetOrThrow_(sheetName);
  const records = getRecords_(sheetName);

  const rowsToDelete = records
    .filter((record) => {
      return ids.has(
        String(record[idColumnName])
      );
    })
    .map((record) => record._rowNumber)
    .sort((firstRow, secondRow) => {
      return secondRow - firstRow;
    });

  rowsToDelete.forEach((rowNumber) => {
    sheet.deleteRow(rowNumber);
  });
}

/**
 * Assigns a real serial number to an anonymous Unbundled PT deal.
 *
 * If no anonymous deal remains, the newly discovered identified deal is added
 * to inventory and an Unexpected Deal Inventory Issue is generated.
 */
function identifyPTDeal_(data) {
  const serialNumber = normalizeText_(
    data.serialNumber
  );

  if (!serialNumber) {
    throw new Error(
      'Serial Number is required.'
    );
  }

  if (
    findPTDeal_(
      data.formNumber,
      serialNumber
    )
  ) {
    throw new Error(
      `Deal ${data.formNumber}/${serialNumber} already exists.`
    );
  }

  const formNumber = normalizeFormNumber_(
    data.formNumber
  );

  const anonymousDeal = getPTDeals_().find((deal) => {
    return (
      String(deal.PT_Game_ID) === String(data.gameId) &&
      normalizeFormNumber_(deal.Form_Number) === formNumber &&
      !normalizeText_(deal.Serial_Number) &&
      String(deal.Deal_Status) === 'Unbundled'
    );
  });

  if (anonymousDeal) {
    updateRecordRow_(
      'PT_Deals',
      anonymousDeal._rowNumber,
      {
        Serial_Number: serialNumber,
        Deal_Status: data.newStatus || 'Bundled',
        Bundled_Date: data.effectiveDate || now_()
      }
    );

    postInventoryTransaction_({
      effectiveDate: data.effectiveDate,
      inventoryType: 'PT',
      productId: data.gameId,
      itemId: anonymousDeal.PT_Deal_ID,
      formNumber: formNumber,
      serialNumber: serialNumber,
      transactionType: 'Bundled',
      quantityChange: 0,
      unit: 'Deal',
      source: data.source || 'PT Processing',
      notes: 'Serial assigned to unidentified deal.'
    });

    return {
      dealId: anonymousDeal.PT_Deal_ID,
      matchedAnonymous: true,
      issueId: ''
    };
  }

  const dealId = createPTDeal_({
    gameId: data.gameId,
    formNumber: formNumber,
    serialNumber: serialNumber,
    status: data.newStatus || 'Bundled',
    receivedDate: data.effectiveDate,
    notes: data.notes
  });

  const transactionId = postInventoryTransaction_({
    effectiveDate: data.effectiveDate,
    inventoryType: 'PT',
    productId: data.gameId,
    itemId: dealId,
    formNumber: formNumber,
    serialNumber: serialNumber,
    transactionType: 'Adjustment',
    quantityChange: 1,
    unit: 'Deal',
    source: data.source || 'PT Processing',
    notes: 'Unexpected deal found during PT processing.'
  });

  const issueId = createInventoryIssue_({
    inventoryType: 'PT',
    issueType: 'Unexpected Deal',
    productId: data.gameId,
    itemId: dealId,
    formNumber: formNumber,
    serialNumber: serialNumber,
    expectedQuantity: 0,
    reportedQuantity: 1,
    discrepancyAmount: 1,
    notes:
      'No unidentified deal was available. ' +
      `Transaction ${transactionId}.`
  });

  return {
    dealId: dealId,
    matchedAnonymous: false,
    issueId: issueId
  };
}

/**
 * Ordering.gs
 * LBA Bingo Inventory Management v1.4
 *
 * Records Director-entered inventory orders.
 *
 * V1.4 changes:
 * - Deals per Case is authoritative from PT_Catalog.
 * - The browser no longer supplies or controls Deals per Case.
 * - The entire order is validated before any inventory record is created.
 */

function recordPTOrder(token, payload) {
  requireAdmin_(token);

  payload = payload || {};

  const gameId = normalizeText_(payload.gameId);
  const formNumber = normalizeFormNumber_(payload.formNumber);

  const catalogGame = getPTCatalog_().find((game) => {
    return String(game.PT_Game_ID) === gameId;
  });

  if (!catalogGame) {
    throw new Error(
      `PT Game ${gameId} was not found in PT_Catalog.`
    );
  }

  const catalogDealsPerCase = integerOrThrow_(
    catalogGame.Deals_Per_Case,
    'Catalog Deals per Case',
    { allowZero: false }
  );

  const validForms = [
    normalizeFormNumber_(catalogGame.Primary_Form_Number),
    ...splitAlternateForms_(catalogGame.Alternate_Form_Numbers)
  ];

  if (!validForms.includes(formNumber)) {
    throw new Error(
      `Form ${formNumber} is not assigned to ${catalogGame.Game_Name}.`
    );
  }

  const casesOrdered = integerOrThrow_(
    payload.cases,
    'Cases Ordered'
  );

  const looseDeals = integerOrThrow_(
    payload.looseDeals || 0,
    'Loose Deals'
  );

  const quantity =
    (casesOrdered * catalogDealsPerCase) +
    looseDeals;

  if (quantity <= 0) {
    throw new Error(
      'The order must add at least one PT deal.'
    );
  }

  const effectiveDate = payload.effectiveDate
    ? new Date(`${payload.effectiveDate}T12:00:00`)
    : now_();

  const notes = normalizeText_(payload.notes) ||
    `${casesOrdered} case(s) and ${looseDeals} loose deal(s) ordered.`;

  const lock = LockService.getDocumentLock();
  lock.waitLock(30000);

  try {
    const dealIds = createAnonymousPTStock_({
      gameId: gameId,
      formNumber: formNumber,
      quantity: quantity,
      status: 'Unbundled',
      effectiveDate: effectiveDate,
      transactionType: 'Received',
      source: 'Inventory Ordering',
      notes: notes
    });

    return {
      quantity: quantity,
      casesOrdered: casesOrdered,
      looseDeals: looseDeals,
      dealsPerCase: catalogDealsPerCase,
      dealIds: dealIds
    };
  } finally {
    lock.releaseLock();
  }
}

const PAPER_CASE_STATUSES = [
  'Unopened',
  'Opened',
  'Expended',
  'Missing'
];


/**
 * Derives the required physical status from a reconciled pack count.
 * Missing remains an explicit administrator choice. An already-opened case
 * does not become Unopened merely because a correction restores full quantity.
 */
function derivePaperCaseStatus_(
  requestedStatus,
  recordedStatus,
  observedPacks,
  packsPerCase
) {
  if (requestedStatus === 'Missing') {
    return 'Missing';
  }
  if (observedPacks === 0) {
    return 'Expended';
  }
  if (observedPacks < packsPerCase) {
    return 'Opened';
  }
  if (recordedStatus === 'Opened') {
    return 'Opened';
  }
  return requestedStatus;
}


function getPaperPacksPerCase_(paperId) {
  const paper = getPaperCatalog_().find((catalogRow) => {
    return String(catalogRow.Paper_ID) === String(paperId);
  });
  if (!paper) {
    throw new Error(`Paper type ${paperId} was not found.`);
  }
  return integerOrThrow_(
    paper.Packs_Per_Case,
    'Packs per Case',
    { allowZero: false }
  );
}


function getPaperReconciliationBootstrap(token) {
  requireAdmin_(token);

  const paperTypes = getPaperCatalog_()
    .filter((paper) => {
      return (
        paper.Active === true ||
        String(paper.Active).toLowerCase() === 'true'
      );
    })
    .map((paper) => {
      return {
        paperId: String(paper.Paper_ID),
        shortName: String(paper.Short_Name),
        description: String(paper.Description),
        packsPerCase: Number(paper.Packs_Per_Case),
        label: `${paper.Short_Name} - ${paper.Description}`
      };
    })
    .sort((first, second) => {
      return first.label.localeCompare(second.label);
    });

  return {
    paperTypes: paperTypes,
    statuses: PAPER_CASE_STATUSES
  };
}


function getPaperCasesForReconciliation(token, paperId) {
  requireAdmin_(token);

  return getRecords_('Paper_Cases')
    .filter((paperCase) => {
      return String(paperCase.Paper_ID) === String(paperId);
    })
    .map((paperCase) => {
      return {
        caseId: String(paperCase.Paper_Case_ID),
        serialNumber: normalizeText_(paperCase.Serial_Number),
        recordedStatus: String(paperCase.Case_Status),
        calculatedPacks: getPaperCasePackBalance_(paperCase.Paper_Case_ID),
        notes: normalizeText_(paperCase.Notes)
      };
    })
    .sort((first, second) => {
      const firstKey = first.serialNumber || first.caseId;
      const secondKey = second.serialNumber || second.caseId;

      return firstKey.localeCompare(
        secondKey,
        undefined,
        { numeric: true }
      );
    });
}


function savePaperReconciliation(token, payload) {
  requireAdmin_(token);

  payload = payload || {};

  const paperId = normalizeText_(payload.paperId);
  const observations = Array.isArray(payload.cases)
    ? payload.cases
    : [];

  if (!paperId) {
    throw new Error('Select a Paper Type.');
  }

  if (!observations.length) {
    throw new Error('No paper-case observations were entered.');
  }

  const effectiveDate = payload.effectiveDate
    ? new Date(`${payload.effectiveDate}T12:00:00`)
    : now_();

  validatePaperReconciliationObservations_(
    paperId,
    observations
  );

  const lock = LockService.getDocumentLock();
  lock.waitLock(30000);

  try {
    const results = observations.map((observation) => {
      if (normalizeText_(observation.caseId)) {
        return reconcileExistingPaperCase_({
          paperId: paperId,
          effectiveDate: effectiveDate,
          observation: observation
        });
      }

      return reconcileNewlyDiscoveredPaperCase_({
        paperId: paperId,
        effectiveDate: effectiveDate,
        observation: observation
      });
    });

    return {
      saved: results.filter((result) => result.result === 'Saved').length,
      unchanged: results.filter((result) => result.result === 'No change').length,
      results: results
    };
  } finally {
    lock.releaseLock();
  }
}


function validatePaperReconciliationObservations_(
  paperId,
  observations
) {
  const submittedCaseIds = new Set();
  const submittedSerials = new Set();

  observations.forEach((observation, index) => {
    const displayRow = index + 1;
    const caseId = normalizeText_(observation.caseId);
    const serialNumber = normalizeText_(observation.serialNumber);
    const observedStatus = normalizeText_(observation.observedStatus);
    const observedPacks = Number(observation.observedPacks);

    if (!PAPER_CASE_STATUSES.includes(observedStatus)) {
      throw new Error(
        `Row ${displayRow} has an invalid case status.`
      );
    }

    if (
      !Number.isInteger(observedPacks) ||
      observedPacks < 0
    ) {
      throw new Error(
        `Row ${displayRow} Physical Packs must be a nonnegative whole number.`
      );
    }

    if (caseId) {
      if (submittedCaseIds.has(caseId)) {
        throw new Error(
          `Paper case ${caseId} appears more than once.`
        );
      }

      submittedCaseIds.add(caseId);

      const existingCase = findPaperCaseById_(caseId);

      if (!existingCase) {
        throw new Error(
          `Paper case ${caseId} could not be found.`
        );
      }

      if (String(existingCase.Paper_ID) !== String(paperId)) {
        throw new Error(
          `Paper case ${caseId} does not belong to the selected Paper Type.`
        );
      }
    } else if (!serialNumber) {
      throw new Error(
        `New physical case on row ${displayRow} requires a serial number.`
      );
    }

    if (serialNumber) {
      const serialKey = `${paperId}::${serialNumber}`;

      if (submittedSerials.has(serialKey)) {
        throw new Error(
          `Serial ${serialNumber} appears more than once in the reconciliation.`
        );
      }

      submittedSerials.add(serialKey);

      const conflict = findConflictingPaperSerial_(
        paperId,
        serialNumber,
        caseId
      );

      if (conflict) {
        throw new Error(
          `Serial ${serialNumber} is already assigned to ${conflict.Paper_Case_ID}.`
        );
      }
    }
  });
}


function reconcileExistingPaperCase_(context) {
  const observation = context.observation;
  const caseId = normalizeText_(observation.caseId);
  const existingCase = findPaperCaseById_(caseId);

  if (!existingCase) {
    throw new Error(`Paper case ${caseId} could not be found.`);
  }

  const recordedSerial = normalizeText_(existingCase.Serial_Number);
  const observedSerial = normalizeText_(observation.serialNumber);
  const recordedStatus = String(existingCase.Case_Status);
  const requestedStatus = normalizeText_(observation.observedStatus);
  const expectedPacks = getPaperCasePackBalance_(caseId);
  const observedPacks = Number(observation.observedPacks);
  const packsPerCase = getPaperPacksPerCase_(context.paperId);
  const observedStatus = derivePaperCaseStatus_(
    requestedStatus,
    recordedStatus,
    observedPacks,
    packsPerCase
  );
  const newNote = normalizeText_(observation.notes);

  const serialChanged = recordedSerial !== observedSerial;
  const statusChanged = recordedStatus !== observedStatus;
  const quantityChanged = expectedPacks !== observedPacks;
  const noteAdded = Boolean(newNote);

  if (
    !serialChanged &&
    !statusChanged &&
    !quantityChanged &&
    !noteAdded
  ) {
    return {
      caseId: caseId,
      result: 'No change'
    };
  }

  const eventNotes = buildPaperReconciliationNotes_(
    recordedSerial,
    observedSerial,
    recordedStatus,
    observedStatus,
    expectedPacks,
    observedPacks,
    newNote
  );

  const reconciliation = createReconciliationRecord_({
    inventoryType: 'Paper',
    productId: context.paperId,
    itemId: caseId,
    formNumber: '',
    serialNumber: observedSerial,
    reconciliationType: 'Inventory Check',
    expectedQuantity: expectedPacks,
    observedQuantity: observedPacks,
    observedPTStatus: '',
    issueId: observation.issueId || '',
    notes: eventNotes
  });

  const updates = {};

  if (serialChanged) {
    updates.Serial_Number = observedSerial;
  }

  if (statusChanged) {
    updates.Case_Status = observedStatus;
  }

  if (
    observedStatus === 'Opened' &&
    !existingCase.Opened_Date
  ) {
    updates.Opened_Date = context.effectiveDate;
  }

  if (
    observedStatus === 'Expended' &&
    !existingCase.Expended_Date
  ) {
    updates.Expended_Date = context.effectiveDate;
  }

  if (noteAdded) {
    updates.Notes = appendInventoryNote_(
      existingCase.Notes,
      newNote
    );
  }

  if (Object.keys(updates).length) {
    updateRecordRow_(
      'Paper_Cases',
      existingCase._rowNumber,
      updates
    );
  }

  let transactionId = '';

  if (reconciliation.correction !== 0) {
    transactionId = postInventoryTransaction_({
      effectiveDate: context.effectiveDate,
      inventoryType: 'Paper',
      productId: context.paperId,
      itemId: caseId,
      formNumber: '',
      serialNumber: observedSerial,
      transactionType: 'Adjustment',
      quantityChange: reconciliation.correction,
      unit: 'Pack',
      source: 'Inventory Reconciliation',
      reconciliationId: reconciliation.id,
      notes: eventNotes
    });

    linkReconciliationTransaction_(
      reconciliation.id,
      transactionId
    );
  }

  return {
    caseId: caseId,
    reconciliationId: reconciliation.id,
    transactionId: transactionId,
    result: 'Saved'
  };
}


function reconcileNewlyDiscoveredPaperCase_(context) {
  const observation = context.observation;
  const serialNumber = normalizeText_(observation.serialNumber);
  const observedPacks = Number(observation.observedPacks);
  const requestedStatus = normalizeText_(observation.observedStatus);
  const packsPerCase = getPaperPacksPerCase_(context.paperId);
  const observedStatus = derivePaperCaseStatus_(
    requestedStatus,
    '',
    observedPacks,
    packsPerCase
  );
  const notes = normalizeText_(observation.notes) ||
    'New physical Paper case discovered during inventory reconciliation.';

  if (!serialNumber) {
    throw new Error(
      'A newly discovered physical Paper case requires a serial number.'
    );
  }

  const caseId = createPaperCase_({
    paperId: context.paperId,
    serialNumber: serialNumber,
    status: observedStatus,
    receivedDate: context.effectiveDate,
    notes: notes
  });

  const reconciliation = createReconciliationRecord_({
    inventoryType: 'Paper',
    productId: context.paperId,
    itemId: caseId,
    formNumber: '',
    serialNumber: serialNumber,
    reconciliationType: 'Inventory Check',
    expectedQuantity: 0,
    observedQuantity: observedPacks,
    observedPTStatus: '',
    issueId: observation.issueId || '',
    notes: notes
  });

  let transactionId = '';

  if (reconciliation.correction !== 0) {
    transactionId = postInventoryTransaction_({
      effectiveDate: context.effectiveDate,
      inventoryType: 'Paper',
      productId: context.paperId,
      itemId: caseId,
      formNumber: '',
      serialNumber: serialNumber,
      transactionType: 'Adjustment',
      quantityChange: reconciliation.correction,
      unit: 'Pack',
      source: 'Inventory Reconciliation',
      reconciliationId: reconciliation.id,
      notes: notes
    });

    linkReconciliationTransaction_(
      reconciliation.id,
      transactionId
    );
  }

  return {
    caseId: caseId,
    reconciliationId: reconciliation.id,
    transactionId: transactionId,
    result: 'Saved'
  };
}


function buildPaperReconciliationNotes_(
  recordedSerial,
  observedSerial,
  recordedStatus,
  observedStatus,
  expectedPacks,
  observedPacks,
  administratorNote
) {
  const changes = [];

  if (recordedSerial !== observedSerial) {
    changes.push(
      `Serial changed from ${recordedSerial || '[blank]'} ` +
      `to ${observedSerial || '[blank]'}.`
    );
  }

  if (recordedStatus !== observedStatus) {
    changes.push(
      `Status changed from ${recordedStatus} to ${observedStatus}.`
    );
  }

  if (expectedPacks !== observedPacks) {
    changes.push(
      `Pack count changed from ${expectedPacks} to ${observedPacks}.`
    );
  }

  const note = normalizeText_(administratorNote);

  if (note) {
    changes.push(note);
  }

  return changes.join(' ');
}

/**
 * Validates the complete submitted reconciliation before changing inventory.
 */
function validatePTReconciliationObservations_(
  gameId,
  formNumber,
  observations
) {
  const submittedDealIds = new Set();
  const submittedSerials = new Set();

  observations.forEach((observation, index) => {
    const rowNumber = index + 1;
    const dealId = normalizeText_(observation.dealId);
    const serialNumber = normalizeText_(observation.serialNumber);
    const observedStatus = normalizeText_(observation.observedStatus);

    if (!PT_DEAL_STATUSES.includes(observedStatus)) {
      throw new Error(
        `Row ${rowNumber} has an invalid observed status.`
      );
    }

    if (dealId) {
      if (submittedDealIds.has(dealId)) {
        throw new Error(
          `Deal ${dealId} appears more than once in the reconciliation.`
        );
      }

      submittedDealIds.add(dealId);

      const existingDeal = findPTDealById_(dealId);

      if (!existingDeal) {
        throw new Error(
          `Existing deal ${dealId} could not be found.`
        );
      }

      if (
        String(existingDeal.PT_Game_ID) !== String(gameId) ||
        normalizeFormNumber_(existingDeal.Form_Number) !== formNumber
      ) {
        throw new Error(
          `Deal ${dealId} does not belong to the selected game/form.`
        );
      }
    } else if (!serialNumber) {
      throw new Error(
        `New physical deal on row ${rowNumber} requires a serial number.`
      );
    }

    if (serialNumber) {
      const serialKey = `${formNumber}::${serialNumber}`;

      if (submittedSerials.has(serialKey)) {
        throw new Error(
          `Serial ${serialNumber} appears more than once in the reconciliation.`
        );
      }

      submittedSerials.add(serialKey);

      const conflict = findConflictingPTSerial_(
        formNumber,
        serialNumber,
        dealId
      );

      if (conflict) {
        throw new Error(
          `Serial ${serialNumber} is already assigned to ` +
          `${conflict.PT_Deal_ID}.`
        );
      }
    }
  });
}


/**
 * Appends a new note to an existing semicolon-delimited Notes history.
 *
 * Existing notes are preserved. Blank additions do not alter the history.
 * The new note is appended exactly once using "; " as the separator.
 *
 * @param {*} existingNotes
 * @param {*} newNote
 * @return {string}
 */
function appendPTDealNote_(existingNotes, newNote) {
  const existing = normalizeText_(existingNotes);
  const addition = normalizeText_(newNote);

  if (!addition) {
    return existing;
  }

  if (!existing) {
    return addition;
  }

  return `${existing}; ${addition}`;
}

/** Shared reconciliation record operations. */
function createReconciliationRecord_(data) {
  const expected = integerOrThrow_(data.expectedQuantity, 'Expected Quantity');
  const observed = integerOrThrow_(data.observedQuantity, 'Observed Quantity');
  const correction = observed - expected;
  const id = generateId_('RECONCILIATION', 'REC-', 7);
  appendRecord_('Reconciliation_Records', {
    Reconciliation_ID: id,
    Timestamp: now_(),
    Inventory_Type: data.inventoryType,
    Product_ID: data.productId,
    Item_ID: data.itemId || '',
    Form_Number: data.formNumber || '',
    Serial_Number: data.serialNumber,
    Reconciliation_Type: data.reconciliationType || 'Inventory Check',
    Expected_Quantity: expected,
    Observed_Quantity: observed,
    Correction_Amount: correction,
    Observed_PT_Status: data.observedPTStatus || '',
    Issue_ID: data.issueId || '',
    Correction_Transaction_ID: '',
    Notes: data.notes || '',
  });
  return {id, correction};
}

function linkReconciliationTransaction_(reconciliationId, transactionId) {
  const record = getRecords_('Reconciliation_Records').find(row => String(row.Reconciliation_ID) === String(reconciliationId));
  if (!record) throw new Error(`Reconciliation ${reconciliationId} was not found.`);
  updateRecordRow_('Reconciliation_Records', record._rowNumber, {Correction_Transaction_ID: transactionId});
}

/**

 *
 * This module permanently defines, initializes, and validates the v1 schema.

 * Apps Script editor > Services > + > Google Sheets API > Add.

 * IMPORTANT:

/**
 * Creates the complete v1 inventory workbook schema in the bound spreadsheet.
 * Safe initialization: aborts rather than overwriting any existing schema sheet.
 */
function initializeInventoryWorkbook() {
  assertAdvancedSheetsService_();

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error('No active spreadsheet. Run this function from the bound inventory workbook.');

  const schemaNames = Object.keys(INVENTORY_SCHEMA_V1.sheets);
  const existingSchemaSheets = schemaNames.filter(name => ss.getSheetByName(name));
  if (existingSchemaSheets.length) {
    throw new Error(
      'Initialization stopped. These schema sheets already exist: ' + existingSchemaSheets.join(', ') +
      '. Do not rerun initialization against an existing inventory workbook.'
    );
  }

  const existingSheets = ss.getSheets();
  const reusableBlankSheet = existingSheets.length === 1 && isSheetBlank_(existingSheets[0])
    ? existingSheets[0]
    : null;

  const created = [];
  try {
    schemaNames.forEach((name, index) => {
      let sheet;
      if (index === 0 && reusableBlankSheet) {
        sheet = reusableBlankSheet;
        sheet.setName(name);
      } else {
        sheet = ss.insertSheet(name);
      }
      created.push(sheet);
      initializeOneSheet_(sheet, name, INVENTORY_SCHEMA_V1.sheets[name]);
    });

    ss.setActiveSheet(ss.getSheetByName('Dashboard'));
    SpreadsheetApp.flush();

    createNativeTables_();
    applyCalculatedColumns_();
    SpreadsheetApp.flush();

    const report = validateInventorySchemaV1(false);
    if (!report.valid) {
      throw new Error('Initialization completed but validation failed:\n' + report.errors.join('\n'));
    }

    tryShowSchemaDialog_('LBA Bingo Inventory', 'Schema v1 was initialized and validated successfully.');
    return report;
  } catch (error) {
    throw new Error('Inventory schema initialization failed: ' + error.message);
  }
}

/**
 * Validates the workbook against schema v1 without changing data.
 * @param {boolean=} showDialog Whether to show a summary dialog. Defaults to true.
 * @return {{valid:boolean, version:string, errors:string[], warnings:string[]}}
 */
/**
 * Repairs v1 validation/format rules on an already initialized workbook.
 * This is safe for the validation failure seen after native Table creation;
 * it does not recreate sheets or alter inventory records.
 */
function repairInventorySchemaV1Validation() {
  rebuildNativeTablesWithTypedColumns_();
  restoreMissingV1Settings_();
  applyCalculatedColumns_();
  SpreadsheetApp.flush();
  return validateInventorySchemaV1(false);
}

function validateInventorySchemaV1(showDialog = true) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const errors = [];
  const warnings = [];

  Object.entries(INVENTORY_SCHEMA_V1.sheets).forEach(([sheetName, definition]) => {
    const sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      errors.push(`Missing sheet: ${sheetName}`);
      return;
    }

    if (definition.headers.length) {
      const actual = sheet.getRange(1, 1, 1, definition.headers.length).getDisplayValues()[0];
      definition.headers.forEach((expected, i) => {
        if (actual[i] !== expected) {
          errors.push(`${sheetName}: expected header ${expected} in column ${i + 1}; found ${actual[i] || '(blank)'}.`);
        }
      });

      const extraHeader = sheet.getRange(1, definition.headers.length + 1).getDisplayValue();
      if (extraHeader) warnings.push(`${sheetName}: unexpected extra header after the defined v1 columns: ${extraHeader}.`);

      validateConfiguredColumns_(sheet, sheetName, definition, errors, warnings);
    }
  });

  validateSettings_(ss, errors, warnings);
  validateNativeTables_(ss, errors, warnings);

  const report = {
    valid: errors.length === 0,
    version: INVENTORY_SCHEMA_V1.version,
    errors,
    warnings,
  };

  console.log(JSON.stringify(report, null, 2));

  if (showDialog) {
    const title = report.valid ? 'Schema v1 validation passed' : 'Schema v1 validation failed';
    const body = [
      report.valid ? 'No schema errors were found.' : `${errors.length} error(s) were found.`,
      warnings.length ? `${warnings.length} warning(s) were found.` : 'No warnings were found.',
      '',
      errors.slice(0, 12).join('\n'),
      errors.length > 12 ? `\n...and ${errors.length - 12} more error(s). See the execution log.` : '',
      warnings.length ? '\nWarnings:\n' + warnings.slice(0, 8).join('\n') : '',
    ].join('\n');
    tryShowSchemaDialog_(title, body);
  }

  return report;
}

function initializeOneSheet_(sheet, sheetName, definition) {
  sheet.clear();
  sheet.clearConditionalFormatRules();

  if (sheetName === 'Dashboard') {
    sheet.getRange('A1').setValue('LBA Bingo Inventory Dashboard')
      .setFontSize(18).setFontWeight('bold').setFontColor('#ffffff').setBackground('#1f4e78');
    sheet.getRange('A3').setValue('Dashboard formulas and layout will be added after catalog and opening inventory data are established.')
      .setFontStyle('italic').setFontColor('#666666');
    sheet.setColumnWidth(1, 820);
    sheet.setFrozenRows(1);
    sheet.setTabColor('#1f4e78');
    return;
  }

  const headers = definition.headers;
  ensureSheetSize_(sheet, INVENTORY_SCHEMA_V1.tableRows, Math.max(headers.length, 3));
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  sheet.getRange(1, 1, 1, headers.length)
    .setFontWeight('bold')
    .setFontColor('#ffffff')
    .setBackground('#1f4e78')
    .setHorizontalAlignment('center')
    .setWrap(true);
  sheet.setFrozenRows(1);
  sheet.setTabColor(sheetName.includes('Catalog') ? '#6aa84f' : sheetName === 'Settings' ? '#999999' : '#3d85c6');

  definition.widths.forEach((width, i) => sheet.setColumnWidth(i + 1, width));
  sheet.getRange(2, 1, INVENTORY_SCHEMA_V1.tableRows - 1, headers.length).setVerticalAlignment('middle');

  applyDefinitionValidation_(sheet, definition);
  applyDefinitionFormats_(sheet, definition);

  if (definition.initialRows && definition.initialRows.length) {
    sheet.getRange(2, 1, definition.initialRows.length, definition.headers.length).setValues(definition.initialRows);
  }
}

function applyDefinitionValidation_(sheet, definition) {
  const totalDataRows = INVENTORY_SCHEMA_V1.tableRows - 1;

  Object.entries(definition.dropdowns || {}).forEach(([header, options]) => {
    const col = columnIndex_(definition.headers, header);
    if (!col) return;
    const rule = SpreadsheetApp.newDataValidation()
      .requireValueInList(options, true)
      .setAllowInvalid(false)
      .setHelpText(`Choose one of: ${options.join(', ')}`)
      .build();
    sheet.getRange(2, col, totalDataRows, 1).setDataValidation(rule);
  });

  (definition.checkboxes || []).forEach(header => {
    const col = columnIndex_(definition.headers, header);
    if (col) sheet.getRange(2, col, totalDataRows, 1).insertCheckboxes();
  });

  (definition.integers || []).forEach(header => {
    const col = columnIndex_(definition.headers, header);
    if (!col) return;
    const range = sheet.getRange(2, col, totalDataRows, 1);
    const rule = SpreadsheetApp.newDataValidation()
      .requireFormulaSatisfied(`=OR(${range.getCell(1, 1).getA1Notation()}="",AND(ISNUMBER(${range.getCell(1, 1).getA1Notation()}),MOD(${range.getCell(1, 1).getA1Notation()},1)=0))`)
      .setAllowInvalid(false)
      .setHelpText('Enter a whole number. Transaction quantities must always be integers.')
      .build();
    range.setDataValidation(rule);
  });
}

function applyDefinitionFormats_(sheet, definition) {
  const rows = INVENTORY_SCHEMA_V1.tableRows - 1;
  formatColumns_(sheet, definition, definition.integers || [], '0', rows);
  formatColumns_(sheet, definition, definition.decimals || [], '0.0###', rows);
  formatColumns_(sheet, definition, definition.currency || [], '$#,##0.00', rows);
  formatColumns_(sheet, definition, definition.percent || [], '0.00%', rows);
  formatColumns_(sheet, definition, definition.dates || [], 'm/d/yyyy', rows);
  formatColumns_(sheet, definition, definition.timestamps || [], 'm/d/yyyy h:mm:ss am/pm', rows);
}

function formatColumns_(sheet, definition, headers, format, rows) {
  headers.forEach(header => {
    const col = columnIndex_(definition.headers, header);
    if (col) sheet.getRange(2, col, rows, 1).setNumberFormat(format);
  });
}

function applyCalculatedColumns_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  Object.entries(INVENTORY_SCHEMA_V1.sheets).forEach(([sheetName, definition]) => {
    if (!definition.formulas) return;
    const sheet = ss.getSheetByName(sheetName);
    Object.entries(definition.formulas).forEach(([header, formula]) => {
      const col = columnIndex_(definition.headers, header);
      if (col) sheet.getRange(2, col).setFormula(formula);
    });
  });
}

function createNativeTables_() {
  addNativeTablesWithTypedColumns_(false);
}

function rebuildNativeTablesWithTypedColumns_() {
  addNativeTablesWithTypedColumns_(true);
}

function addNativeTablesWithTypedColumns_(replaceExisting) {
  assertAdvancedSheetsService_();
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  if (replaceExisting) {
    const current = Sheets.Spreadsheets.get(ss.getId(), {includeGridData: false});
    const deleteRequests = [];
    (current.sheets || []).forEach(apiSheet => {
      (apiSheet.tables || []).forEach(table => {
        deleteRequests.push({deleteTable: {tableId: table.tableId}});
      });
    });
    if (deleteRequests.length) {
      Sheets.Spreadsheets.batchUpdate({requests: deleteRequests}, ss.getId());
    }
  }

  const requests = [];
  Object.entries(INVENTORY_SCHEMA_V1.sheets).forEach(([sheetName, definition]) => {
    if (!definition.tableName || !definition.headers.length) return;
    const sheet = ss.getSheetByName(sheetName);
    const rowCount = Math.max(definition.initialRows ? definition.initialRows.length + 2 : 20, 20);
    const columnProperties = buildTableColumnProperties_(definition);
    requests.push({
      addTable: {
        table: {
          name: definition.tableName,
          range: {
            sheetId: sheet.getSheetId(),
            startRowIndex: 0,
            endRowIndex: rowCount,
            startColumnIndex: 0,
            endColumnIndex: definition.headers.length,
          },
          columnProperties: columnProperties,
        },
      },
    });
  });
  if (requests.length) Sheets.Spreadsheets.batchUpdate({requests}, ss.getId());
}

function buildTableColumnProperties_(definition) {
  return definition.headers.map((header, zeroBasedIndex) => {
    const property = {
      columnIndex: zeroBasedIndex,
      columnName: header,
    };
    if ((definition.dropdowns || {})[header]) {
      property.columnType = 'DROPDOWN';
      property.dataValidationRule = {
        condition: {
          type: 'ONE_OF_LIST',
          values: definition.dropdowns[header].map(value => ({userEnteredValue: value})),
        },
      };
    } else if ((definition.checkboxes || []).includes(header)) {
      // The Sheets REST API does not currently accept CHECKBOX as a
      // TableColumnProperties ColumnType. Leave the native table column
      // untyped, then ordinary checkbox validation is applied only to
      // cells OUTSIDE the native table range by Apps Script. For Active
      // catalog fields, TRUE/FALSE values remain valid in the table itself.
    }
    return property;
  });
}

function validateConfiguredColumns_(sheet, sheetName, definition, errors, warnings) {
  Object.keys(definition.formulas || {}).forEach(header => {
    const col = columnIndex_(definition.headers, header);
    if (!sheet.getRange(2, col).getFormula()) {
      errors.push(`${sheetName}.${header}: missing calculated-column formula in row 2.`);
    }
  });
}

function restoreMissingV1Settings_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Settings');
  if (!sheet) throw new Error('Missing required sheet: Settings');

  const lastRow = sheet.getLastRow();
  const existingValues = lastRow > 1
    ? sheet.getRange(2, 1, lastRow - 1, 3).getValues()
    : [];
  const existing = new Set(existingValues.filter(row => row[0]).map(row => String(row[0])));
  const missing = INVENTORY_SCHEMA_V1.sheets.Settings.initialRows.filter(row => !existing.has(row[0]));

  if (missing.length) {
    const startRow = Math.max(sheet.getLastRow() + 1, 2);
    sheet.getRange(startRow, 1, missing.length, 3).setValues(missing);
  }
}

function validateSettings_(ss, errors, warnings) {
  const sheet = ss.getSheetByName('Settings');
  if (!sheet) return;

  const dataRowCount = Math.max(0, sheet.getLastRow() - 1);
  const values = dataRowCount > 0
    ? sheet.getRange(2, 1, dataRowCount, 2).getDisplayValues()
    : [];
  const settings = Object.fromEntries(values.filter(row => row[0]).map(row => [row[0], row[1]]));

  const required = INVENTORY_SCHEMA_V1.sheets.Settings.initialRows.map(row => row[0]);
  required.forEach(key => {
    if (!(key in settings)) errors.push(`Settings: missing required setting ${key}.`);
  });

  if (settings.Schema_Version !== INVENTORY_SCHEMA_V1.version) {
    errors.push(`Settings.Schema_Version: expected ${INVENTORY_SCHEMA_V1.version}; found ${settings.Schema_Version || '(blank)'}.`);
  }

  ['Inventory_Start_Date', 'Thursday_Paperwork_Folder_ID', 'Friday_Paperwork_Folder_ID', 'Sunday_Paperwork_Folder_ID', 'Paperwork_Template_Folder_ID']
    .forEach(key => {
      if (key in settings && !settings[key]) warnings.push(`Settings.${key} is currently blank.`);
    });
}

function validateNativeTables_(ss, errors, warnings) {
  try {
    assertAdvancedSheetsService_();
    const response = Sheets.Spreadsheets.get(ss.getId(), {includeGridData: false});
    const found = new Set();
    (response.sheets || []).forEach(apiSheet => {
      (apiSheet.tables || []).forEach(table => found.add(table.name));
    });

    Object.values(INVENTORY_SCHEMA_V1.sheets).forEach(definition => {
      if (definition.tableName && !found.has(definition.tableName)) {
        errors.push(`Missing native Google Sheets table: ${definition.tableName}.`);
      }
    });
  } catch (error) {
    warnings.push('Native table validation could not run: ' + error.message);
  }
}

function tryShowSchemaDialog_(title, body) {
  try {
    const ui = SpreadsheetApp.getUi();
    ui.alert(title, body, ui.ButtonSet.OK);
  } catch (error) {
    console.log(`${title}: ${body}`);
  }
}

function assertAdvancedSheetsService_() {
  if (typeof Sheets === 'undefined' || !Sheets.Spreadsheets) {
    throw new Error(
      'The Advanced Google Sheets service is not enabled. In Apps Script, open Services, click +, add Google Sheets API, and rerun.'
    );
  }
}

function columnIndex_(headers, header) {
  const index = headers.indexOf(header);
  return index < 0 ? 0 : index + 1;
}

function ensureSheetSize_(sheet, rows, columns) {
  if (sheet.getMaxRows() < rows) sheet.insertRowsAfter(sheet.getMaxRows(), rows - sheet.getMaxRows());
  if (sheet.getMaxColumns() < columns) sheet.insertColumnsAfter(sheet.getMaxColumns(), columns - sheet.getMaxColumns());
}

function isSheetBlank_(sheet) {
  return sheet.getLastRow() === 1 && sheet.getLastColumn() === 1 && sheet.getRange('A1').isBlank();
}

/**
 * Pre-production schema patch for Inventory Management v1.2.
 *
 * This version is safe when a previous patch attempt removed the contents of
 * Transactions or Inventory_Issues while replacing their native Sheets tables.
 *
 * Run applySchemaV12Patch() once before recording inventory transactions.
 */

function applySchemaV12Patch() {
  ensureV12LedgerHeaders_();

  const transactions = getRecords_('Transactions');

  if (transactions.length > 0) {
    throw new Error(
      'The V1.2 pre-production patch requires an empty Transactions sheet.'
    );
  }

  ensureApplicationSettings();

  upsertV12Setting_(
    'App_Version',
    '1.2',
    'Current integrated application version.'
  );

  rebuildV12TypedTables_();

  return {
    ok: true,
    version: '1.2'
  };
}


/**
 * Restores the two table headers touched by the V1.2 patch if a previous
 * failed patch attempt left either sheet blank.
 *
 * This is intentionally limited to empty sheets so existing data cannot be
 * overwritten accidentally.
 */
function ensureV12LedgerHeaders_() {
  const requiredHeaders = {
    Transactions: [
      'Transaction_ID',
      'Effective_Date',
      'Timestamp',
      'Inventory_Type',
      'Product_ID',
      'Item_ID',
      'Form_Number',
      'Serial_Number',
      'Transaction_Type',
      'Quantity_Change',
      'Unit',
      'Session_ID',
      'Source',
      'Reconciliation_ID',
      'Notes'
    ],

    Inventory_Issues: [
      'Issue_ID',
      'Timestamp',
      'Inventory_Type',
      'Issue_Type',
      'Product_ID',
      'Item_ID',
      'Form_Number',
      'Serial_Number',
      'Session_ID',
      'Expected_Quantity',
      'Reported_Quantity',
      'Discrepancy_Amount',
      'Status',
      'Reconciliation_ID',
      'Notes'
    ]
  };

  Object.entries(requiredHeaders).forEach(([sheetName, headers]) => {
    const sheet = getSheetOrThrow_(sheetName);

    if (sheet.getLastColumn() === 0 || sheet.getLastRow() === 0) {
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      return;
    }

    const existingHeaders = sheet
      .getRange(1, 1, 1, Math.min(sheet.getLastColumn(), headers.length))
      .getDisplayValues()[0];

    const sheetHasContent = existingHeaders.some((value) => value !== '');

    if (!sheetHasContent) {
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    }
  });

  SpreadsheetApp.flush();
}


function upsertV12Setting_(name, value, description) {
  const existing = getRecords_('Settings').find(
    (record) => String(record.Setting) === name
  );

  if (existing) {
    updateRecordRow_('Settings', existing._rowNumber, {
      Value: value,
      Description: description
    });
    return;
  }

  appendRecord_('Settings', {
    Setting: name,
    Value: value,
    Description: description
  });
}


/**
 * Rebuilds only the Transactions and Inventory_Issues native Tables.
 *
 * Google Sheets DeleteTableRequest deletes the table contents as well as the
 * table object. Therefore this routine captures the header row BEFORE deleting
 * each table, immediately restores the headers, and only operates while the
 * pre-production Transactions ledger is empty.
 */
function rebuildV12TypedTables_() {
  const spreadsheet = getInventorySpreadsheet_();

  const tableDefinitions = {
    Transactions_Table: {
      sheetName: 'Transactions',
      dropdowns: {
        Inventory_Type: ['PT', 'Paper'],
        Transaction_Type: [
          'Initial Inventory',
          'Received',
          'Bundled',
          'Opened',
          'Sold',
          'Adjustment',
          'Returned',
          'Destroyed'
        ],
        Unit: ['Deal', 'Pack'],
        Source: [
          'Inventory Reconciliation',
          'Inventory Ordering',
          'Session Paperwork Processing',
          'PT Processing',
          'Paper Processing',
          'Admin Manual'
        ]
      }
    },

    Inventory_Issues_Table: {
      sheetName: 'Inventory_Issues',
      dropdowns: {
        Inventory_Type: ['PT', 'Paper'],
        Issue_Type: [
          'Unknown Serial',
          'Insufficient Inventory',
          'Paper Type Mismatch',
          'Unknown Deal',
          'Unexpected Deal',
          'Duplicate Sale',
          'Form Mismatch',
          'Unavailable Deal'
        ],
        Status: ['Open', 'Resolved', 'Dismissed']
      }
    }
  };

  const apiSpreadsheet = Sheets.Spreadsheets.get(
    spreadsheet.getId(),
    { includeGridData: false }
  );

  Object.entries(tableDefinitions).forEach(([tableName, definition]) => {
    rebuildOneV12Table_(
      spreadsheet,
      apiSpreadsheet,
      tableName,
      definition
    );
  });
}


function rebuildOneV12Table_(
  spreadsheet,
  apiSpreadsheet,
  tableName,
  definition
) {
  const sheet = getSheetOrThrow_(definition.sheetName);
  const columnCount = sheet.getLastColumn();

  if (columnCount < 1) {
    throw new Error(
      `${definition.sheetName} has no header columns. ` +
      'Run ensureV12LedgerHeaders_() before rebuilding the table.'
    );
  }

  const headers = sheet
    .getRange(1, 1, 1, columnCount)
    .getDisplayValues()[0];

  if (headers.some((header) => !header)) {
    throw new Error(
      `${definition.sheetName} contains a blank header cell. ` +
      'The V1.2 table cannot be rebuilt safely.'
    );
  }

  const sheetApi = (apiSpreadsheet.sheets || []).find(
    (candidate) => candidate.properties.sheetId === sheet.getSheetId()
  );

  const existingTable = sheetApi
    ? (sheetApi.tables || []).find((table) => table.name === tableName)
    : null;

  if (existingTable) {
    Sheets.Spreadsheets.batchUpdate(
      {
        requests: [
          {
            deleteTable: {
              tableId: existingTable.tableId
            }
          }
        ]
      },
      spreadsheet.getId()
    );

    // DeleteTableRequest removes the table data as well. Restore the captured
    // header row immediately before creating the replacement table.
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    SpreadsheetApp.flush();
  }

  const columnProperties = headers.map((header, columnIndex) => {
    const property = {
      columnIndex: columnIndex,
      columnName: header
    };

    const dropdownValues = definition.dropdowns[header];

    if (dropdownValues) {
      property.columnType = 'DROPDOWN';
      property.dataValidationRule = {
        condition: {
          type: 'ONE_OF_LIST',
          values: dropdownValues.map((dropdownValue) => ({
            userEnteredValue: dropdownValue
          }))
        }
      };
    }

    return property;
  });

  Sheets.Spreadsheets.batchUpdate(
    {
      requests: [
        {
          addTable: {
            table: {
              name: tableName,
              range: {
                sheetId: sheet.getSheetId(),
                startRowIndex: 0,
                endRowIndex: 20,
                startColumnIndex: 0,
                endColumnIndex: headers.length
              },
              columnProperties: columnProperties
            }
          }
        }
      ]
    },
    spreadsheet.getId()
  );
}

/**
 * TransactionFormatting.gs
 * LBA Bingo Inventory Management v1.10
 *
 * Ensures Transactions.Effective_Date displays as a date and
 * Transactions.Timestamp displays full date and time.
 */

function applyTransactionTimestampFormattingV19() {
  const sheet = getSheetOrThrow_('Transactions');
  const headerMap = getHeaderMap_(sheet);

  if (!('Effective_Date' in headerMap)) {
    throw new Error('Transactions.Effective_Date column was not found.');
  }

  if (!('Timestamp' in headerMap)) {
    throw new Error('Transactions.Timestamp column was not found.');
  }

  const dataRows = Math.max(
    sheet.getMaxRows() - 1,
    1
  );

  sheet
    .getRange(
      2,
      headerMap.Effective_Date + 1,
      dataRows,
      1
    )
    .setNumberFormat('ddd mmm dd');

  sheet
    .getRange(
      2,
      headerMap.Timestamp + 1,
      dataRows,
      1
    )
    .setNumberFormat('ddd mmm dd HH:mm:ss');

  return {
    ok: true,
    effectiveDateFormat: 'ddd mmm dd',
    timestampFormat: 'ddd mmm dd HH:mm:ss'
  };
}



/** Version-specific entry point for applying the V1.10 display formats. */
function applyTransactionTimestampFormattingV110() {
  return applyTransactionTimestampFormattingV19();
}

  function updateSuggestedPaperStatus(row) {
    const selectedPaper = selectedPaperType();
    const packsInput = row.querySelector('.physical-packs');
    const statusSelect = row.querySelector('.paper-observed-status');
    const observedPacks = Number(packsInput.value);

    if (
      !selectedPaper ||
      !Number.isInteger(observedPacks) ||
      observedPacks < 0 ||
      statusSelect.value === 'Missing'
    ) {
      return;
    }

    if (observedPacks === 0) {
      statusSelect.value = 'Expended';
      return;
    }

    if (observedPacks < Number(selectedPaper.packsPerCase)) {
      statusSelect.value = 'Opened';
      return;
    }

    if (row.dataset.recordedStatus !== 'Opened') {
      statusSelect.value = 'Unopened';
    }
  }
