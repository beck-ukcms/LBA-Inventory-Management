/**
 * Admin.js
 * Administrative authentication and protected endpoints.
 * Scope: Creates, validates, refreshes, and ends administrator sessions; exposes protected maintenance entry points.
 * Current as of v1.20.
 */
const ADMIN_TOKEN_TTL_SECONDS = 1800;

function authenticateAdmin(code) {
  const expected = PropertiesService.getScriptProperties().getProperty('ADMIN_CODE');
  if (!expected) throw new Error('ADMIN_CODE has not been configured in Script Properties.');
  if (String(code || '') !== String(expected)) return {ok: false, message: 'Incorrect administrative code.'};
  const token = Utilities.getUuid();
  CacheService.getScriptCache().put(`ADMIN_TOKEN_${token}`, '1', ADMIN_TOKEN_TTL_SECONDS);
  return {ok: true, token, expiresInMinutes: ADMIN_TOKEN_TTL_SECONDS / 60};
}

function requireAdmin_(token) {
  if (!token || CacheService.getScriptCache().get(`ADMIN_TOKEN_${token}`) !== '1') {
    throw new Error('Administrative session expired. Return to the main page and sign in again.');
  }
  CacheService.getScriptCache().put(`ADMIN_TOKEN_${token}`, '1', ADMIN_TOKEN_TTL_SECONDS);
}

function endAdminSession(token) {
  if (token) CacheService.getScriptCache().remove(`ADMIN_TOKEN_${token}`);
  return true;
}

function runSchemaValidationFromAdmin(token) {
  requireAdmin_(token);
  return validateInventorySchemaV1(false);
}

  function server(functionName, ...argumentsList) {
    return new Promise((resolve, reject) => {
      google.script.run
        .withSuccessHandler(resolve)
        .withFailureHandler((error) => {

        })[functionName](...argumentsList);
    });
  }

  async function withBusyButton(buttonOrId, action) {
      const button = typeof buttonOrId === 'string'
        ? $(buttonOrId)
        : buttonOrId;
      if (typeof action !== 'function') {
        throw new Error('A busy-button action is required.');
      }
      if (!button) {
        return action();
      }
      const wasDisabled = button.disabled;
      button.disabled = true;
      try {
        return await action();
      } finally {
        button.disabled = wasDisabled;
      }
  }

function localIsoDate(date = new Date()) {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const day = String(date.getDate()).padStart(2, '0');

  return `${year}-${month}-${day}`;
}

function escapeHtml(value) {
  return String(value ?? '').replace(
    /[&<>"']/g,
    (character) => ({
      '&': '&amp;',
      '<': '&lt;',
      '>': '&gt;',
      '"': '&quot;',
      "'": '&#39;'
    })[character]
  );
}


  /**
   * Populates a select element using a shared, consistent implementation.
   */
  function populateSelectOptions(
    selectOrId,
    values,
    options = {}
  ) {
    const select = typeof selectOrId === 'string'
      ? $(selectOrId)
      : selectOrId;

    if (!select) {
      return;
    }

    const valueProperty = options.valueProperty || '';
    const labelProperty = options.labelProperty || '';
    const placeholder = options.placeholder || '';

    const optionHtml = [];

    if (placeholder) {
      optionHtml.push(
        `<option value="">${escapeHtml(placeholder)}</option>`
      );
    }

    (values || []).forEach((item, index) => {
      const optionValue = valueProperty
        ? item[valueProperty]
        : (typeof item === 'object' ? index : item);

      const optionLabel = labelProperty
        ? item[labelProperty]
        : (typeof item === 'object' ? String(item) : item);

      optionHtml.push(
        `<option value="${escapeHtml(optionValue)}">` +
        `${escapeHtml(optionLabel)}` +
        '</option>'
      );
    });

    select.innerHTML = optionHtml.join('');
  }

    populateSelectOptions(

      gameTypes || []
    );
  }

  let INVENTORY_ISSUES = [];
  let INVENTORY_ISSUES_FILTER = 'Open';
  let SELECTED_INVENTORY_ISSUE = null;


  async function openInventoryIssues() {
    showScreen('inventory-issues-screen');
    await loadInventoryIssues('Open');

async function loadInventoryIssues(filter) {
  INVENTORY_ISSUES_FILTER = filter;
  clearStatus('inventory-issues-status');
  ['Open', 'Resolved', 'All'].forEach((option) => {
    const button = $(`issues-${option.toLowerCase()}-tab`);
    button.className = option === filter ? 'btn' : 'btn secondary';
  });
  try {
    const issues = await server('getInventoryIssues', ADMIN_TOKEN, filter);
    if (!Array.isArray(issues)) {
      throw new Error('Inventory Issues returned an invalid response. Please reload and try again.');
    }
    INVENTORY_ISSUES = issues;
    renderInventoryIssues();
  } catch (error) {
    INVENTORY_ISSUES = [];
    $('inventory-issues-rows').innerHTML = '';
    setStatus('inventory-issues-status', error.message, true);
  }
}


  function renderInventoryIssues() {
    const body = $('inventory-issues-rows');

    if (!INVENTORY_ISSUES.length) {
      body.innerHTML = `
        <tr>
          <td colspan="10" class="muted">
            No ${escapeHtml(INVENTORY_ISSUES_FILTER.toLowerCase())}
            inventory issues found.
          </td>
        </tr>
      `;
      return;
    }

    body.innerHTML = INVENTORY_ISSUES.map((issue) => {
      const created = issue.timestamp
        ? new Date(issue.timestamp).toLocaleString()
        : '';

      const product = issue.formNumber
        ? `${issue.productLabel} (${issue.formNumber})`
        : issue.productLabel;

      return `
        <tr>
          <td>${escapeHtml(created)}</td>
          <td>${escapeHtml(issue.inventoryType)}</td>
          <td>${escapeHtml(issue.issueType)}</td>
          <td>${escapeHtml(product)}</td>
          <td>${escapeHtml(issue.serialNumber)}</td>
          <td>${issue.expectedQuantity}</td>
          <td>${issue.reportedQuantity}</td>
          <td>${escapeHtml(issue.sessionId)}</td>
          <td>${escapeHtml(issue.status)}</td>
          <td>
            <button
              class="btn secondary"
              type="button"
              onclick="reviewInventoryIssue('${escapeHtml(issue.issueId)}')"
            >
              Review
            </button>
          </td>
        </tr>
      `;
    }).join('');
  }


  function reviewInventoryIssue(issueId) {
    SELECTED_INVENTORY_ISSUE = INVENTORY_ISSUES.find((issue) => {
      return issue.issueId === issueId;
    }) || null;

    if (!SELECTED_INVENTORY_ISSUE) {
      setStatus(
        'inventory-issues-status',
        'The selected Inventory Issue could not be found.',
        true
      );
      return;
    }

    const issue = SELECTED_INVENTORY_ISSUE;
    const rows = [
      ['Issue ID', issue.issueId],
      ['Created', issue.timestamp ? new Date(issue.timestamp).toLocaleString() : ''],
      ['Inventory Type', issue.inventoryType],
      ['Issue Type', issue.issueType],
      ['Product', issue.productLabel],
      ['Product ID', issue.productId],
      ['Item ID', issue.itemId],
      ['Form Number', issue.formNumber],
      ['Serial Number', issue.serialNumber],
      ['Session ID', issue.sessionId],
      ['Expected Quantity', issue.expectedQuantity],
      ['Reported Quantity', issue.reportedQuantity],
      ['Discrepancy', issue.discrepancyAmount],
      ['Status', issue.status],
      ['Reconciliation ID', issue.reconciliationId],
      ['Original Notes', issue.notes],
      ['Resolved At', issue.resolvedAt ? new Date(issue.resolvedAt).toLocaleString() : ''],
      ['Resolution Notes', issue.resolutionNotes]
    ];

    $('inventory-issue-details').innerHTML = `
      <table>
        <tbody>
          ${rows.map((row) => `
            <tr>
              <th>${escapeHtml(row[0])}</th>
              <td>${escapeHtml(row[1])}</td>
            </tr>
          `).join('')}
        </tbody>
      </table>
    `;

    $('inventory-issue-resolution-panel').classList.toggle(
      'hidden',
      issue.status !== 'Open'
    );

    $('inventory-issue-resolution-notes').value = '';
    clearStatus('inventory-issue-review-status');
    openModal('inventory-issue-review-modal');
  }


  async function resolveSelectedInventoryIssue() {
    if (!SELECTED_INVENTORY_ISSUE) {
      return;
    }

    clearStatus('inventory-issue-review-status');

    try {
      await server(
        'resolveInventoryIssue',
        ADMIN_TOKEN,
        SELECTED_INVENTORY_ISSUE.issueId,
        $('inventory-issue-resolution-notes').value
      );

      closeModal('inventory-issue-review-modal');
      await loadInventoryIssues(INVENTORY_ISSUES_FILTER);

      setStatus(
        'inventory-issues-status',
        `Inventory Issue ${SELECTED_INVENTORY_ISSUE.issueId} was marked Resolved.`
      );
    } catch (error) {
      setStatus(
        'inventory-issue-review-status',
        error.message,
        true
      );
    }
  }
</script>

<script>
  const PT_BUNDLING_GAMES = new Map();
  let PT_BUNDLING_PRINTING = false;


  function ptBundlingKey(value) {
    return String(value ?? '')
      .trim()
      .toLowerCase();
  }


  async function openPTBundling() {
    showScreen('pt-bundling-screen');

    PT_BUNDLING_GAMES.clear();
    $('pt-bundle-form').disabled = true;
    $('pt-bundle-serial').disabled = true;
    $('pt-bundle-print').disabled = true;
    $('pt-bundle-match').hidden = true;

    setStatus(
      'pt-bundle-status',
      'Loading games...'
    );

    try {
      const games = await server('getPTBundlingGames');

      games.forEach((game) => {
        const key = ptBundlingKey(game.formNumber);

        if (!PT_BUNDLING_GAMES.has(key)) {
          PT_BUNDLING_GAMES.set(key, game);
        }
      });

      $('pt-bundle-form').disabled = false;
      $('pt-bundle-serial').disabled = false;
      $('pt-bundle-print').disabled = false;

      setStatus(
        'pt-bundle-status',
        `Loaded ${games.length} game/form option(s).`
      );

      $('pt-bundle-form').focus();
    } catch (error) {
      setStatus(
        'pt-bundle-status',
        error.message,
        true
      );
    }
  }


  function closePTBundling() {
    clearPTBundlingInputs();
    showScreen('home-screen');
  }


  function showPTBundlingMatch() {
    const game = PT_BUNDLING_GAMES.get(
      ptBundlingKey($('pt-bundle-form').value)
    );

    if (!game) {
      $('pt-bundle-match').hidden = true;
      return;
    }

    $('pt-bundle-match').hidden = false;
    $('pt-bundle-match').textContent =
      `${game.name} | Cards: ${game.cards} | Payout: ${game.payout}`;
  }


  async function submitPTBundle() {
    const formNumber = $('pt-bundle-form').value.trim();
    const serialNumber = $('pt-bundle-serial').value.trim();

    if (!formNumber || !serialNumber) {
      setStatus(
        'pt-bundle-status',
        'Enter both a Form Number and Serial Number.',
        true
      );

      (!formNumber
        ? $('pt-bundle-form')
        : $('pt-bundle-serial'))
        .focus();

      return;
    }

    const game = PT_BUNDLING_GAMES.get(
      ptBundlingKey(formNumber)
    );

    if (!game) {
      setStatus(
        'pt-bundle-status',
        `Form Number ${formNumber} was not found.`,
        true
      );

      $('pt-bundle-form').focus();
      $('pt-bundle-form').select();
      return;
    }

    $('pt-bundle-print').disabled = true;

    setStatus(
      'pt-bundle-status',
      'Recording bundle in inventory...'
    );

    try {
      const roundTripStartedAt = performance.now();
      const result = await server(
        'bundlePTDeal',
        formNumber,
        serialNumber
      );
      const roundTripMs = Math.round(performance.now() - roundTripStartedAt);

      populatePTBundlingReceipt(result);

      PT_BUNDLING_PRINTING = true;

      setStatus(
        'pt-bundle-status',
        `${result.action}. Inventory ID ${result.dealId}.`
      );

      if (result.performance) {
        const lines = [
          'PT Bundling Performance ' + result.performance.version,
          'Browser round trip: ' + roundTripMs + ' ms',
          'Server measured: ' + result.performance.totalMs + ' ms',
          'RPC / unmeasured: ' + Math.max(0, roundTripMs - result.performance.totalMs) + ' ms',
          '',
          'Server steps:'
        ];
        result.performance.steps.forEach((step) => {
          lines.push('  ' + step.name + ': ' + step.ms + ' ms');
        });
        if (result.performance.append) {
          lines.push('', 'Bundled transaction append: ' + result.performance.append.totalMs + ' ms');
          result.performance.append.steps.forEach((step) => {
            lines.push('  ' + step.name + ': ' + step.ms + ' ms');
          });
        }
        const diagnostic = $('pt-bundle-performance');
        diagnostic.textContent = lines.join('\n');
        diagnostic.style.display = 'block';
      }

      requestAnimationFrame(() => {
        window.print();
      });
    } catch (error) {
      $('pt-bundle-print').disabled = false;

      setStatus(
        'pt-bundle-status',
        error.message,
        true
      );
    }
  }


  function populatePTBundlingReceipt(result) {
    $('pt-r-name').textContent = result.gameName;
    $('pt-r-serial').textContent = result.serialNumber;
    $('pt-r-form').textContent = result.formNumber;
    $('pt-r-cards').textContent = result.cards;
    $('pt-r-deal-id').textContent = result.dealId;
    $('pt-r-date').textContent = new Intl.DateTimeFormat(
      undefined,
      {
        year: 'numeric',
        month: '2-digit',
        day: '2-digit'
      }
    ).format(new Date());
  }


  function clearPTBundlingInputs() {
    $('pt-bundle-form').value = '';
    $('pt-bundle-serial').value = '';
    $('pt-bundle-match').hidden = true;
  }


  function finishPTBundlingPrint() {
    if (!PT_BUNDLING_PRINTING) {
      return;
    }

    PT_BUNDLING_PRINTING = false;
    clearPTBundlingInputs();
    $('pt-bundle-print').disabled = false;

    setStatus(
      'pt-bundle-status',
      'Ready for the next receipt.'
    );

    setTimeout(() => {
      $('pt-bundle-form').focus();
    }, 0);
  }


  $('pt-bundle-form').addEventListener(
    'input',
    showPTBundlingMatch
  );

  ['pt-bundle-form', 'pt-bundle-serial'].forEach((elementId) => {
    $(elementId).addEventListener('keydown', (event) => {
      if (
        event.key === 'Enter' &&
        !$('pt-bundle-print').disabled
      ) {
        submitPTBundle();
      }
    });
  });

  window.addEventListener(
    'afterprint',
    finishPTBundlingPrint
  );
</script>

function buildPTFormOptions_() {
  const options = [];
  getPTCatalog_().filter(isActiveCatalogRecord_).forEach((game) => {
    const gameId = String(game.PT_Game_ID);
    const primaryFormNumber = normalizeFormNumber_(game.Primary_Form_Number);
    getPTGameForms_(game).forEach((formNumber) => {
      options.push({
        gameId: gameId,
        gameName: String(game.Game_Name),
        formNumber: formNumber,
        primaryFormNumber: primaryFormNumber,
        isAlternate: formNumber !== primaryFormNumber,
        label: `${game.Game_Name} (${formNumber})`
      });
    });
  });

    return a.gameName.localeCompare(b.gameName) ||
      a.formNumber.localeCompare(b.formNumber);
  });
}
function assertFormAvailable_(formNumber, excludingGameId) {

  if (!form) throw new Error('Form Number is required.');
  const collision = getPTCatalog_().find(game => {
    if (excludingGameId && String(game.PT_Game_ID) === String(excludingGameId)) return false;

    return [normalizeFormNumber_(game.Primary_Form_Number), ...splitAlternateForms_(game.Alternate_Form_Numbers)].includes(form);
  });

/**

const INVENTORY_STRUCTURE_CACHE_VERSION_ = 'v120-1';

function inventoryStructureCacheKey_(kind, sheet) {

function invalidateInventoryHeaderCache_(sheet) {

function cacheInventoryStructureValue_(key, value) {
  const cache = getInventoryStructureCache_();
  if (!cache) return;
  try {
    cache.put(
      key,
      JSON.stringify(value),
      INVENTORY_STRUCTURE_CACHE_TTL_SECONDS_
    );
  } catch (error) {
    // Cache failure must never block inventory operations.
  }
}

function readInventoryStructureCache_(key) {
  const cache = getInventoryStructureCache_();
  if (!cache) return null;
  try {
    const value = cache.get(key);
    return value ? JSON.parse(value) : null;
  } catch (error) {
    return null;
  }
}

function getSheetHeaders_(sheet) {
  const cacheKey = inventoryStructureCacheKey_('HEADERS', sheet);
  const cached = readInventoryStructureCache_(cacheKey);
  if (
    cached &&
    Array.isArray(cached.headers) &&
    cached.headers.length
  ) {
    return cached.headers.slice();
  }
  const lastColumn = sheet.getLastColumn();
  if (lastColumn < 1) {
    throw new Error(`Sheet ${sheet.getName()} has no headers.`);
  }
  const headers = sheet
    .getRange(1, 1, 1, lastColumn)
    .getDisplayValues()[0];
  cacheInventoryStructureValue_(cacheKey, {
    headers: headers,
    headerMap: Object.fromEntries(
      headers.map((header, index) => [header, index])
    )
  });
  return headers.slice();
}

function getHeaderMap_(sheet) {
  const cacheKey = inventoryStructureCacheKey_('HEADERS', sheet);
  const cached = readInventoryStructureCache_(cacheKey);
  if (cached && cached.headerMap) {
    return Object.assign({}, cached.headerMap);
  }
  const headers = getSheetHeaders_(sheet);
  return Object.fromEntries(
    headers.map((header, index) => [header, index])
  );
}


/**
 * Reads all populated records from a worksheet.
 *
 * The returned objects include _rowNumber so that a record can later
 * be updated without depending on hard-coded row positions.
 *
 * @param {string} sheetName
 * @return {Object[]}
 */
const EXECUTION_RECORD_CACHE_ = Object.create(null);
let EXECUTION_NATIVE_TABLE_CACHE_ = null;
let PT_BUNDLING_APPEND_PERF_ = null;
function invalidateRecordCache_(sheetName) {
  delete EXECUTION_RECORD_CACHE_[sheetName];
}
function getRecords_(sheetName) {
  if (EXECUTION_RECORD_CACHE_[sheetName]) {
    return EXECUTION_RECORD_CACHE_[sheetName].map((record) => {
      return Object.assign({}, record);
    });
  }
  const sheet = getSheetOrThrow_(sheetName);
  const lastRow = sheet.getLastRow();
  const lastColumn = sheet.getLastColumn();
  if (lastRow < 2 || lastColumn < 1) {
    EXECUTION_RECORD_CACHE_[sheetName] = [];
    return [];
  }
  const headers = sheet.getRange(1, 1, 1, lastColumn).getDisplayValues()[0];
  const values = sheet.getRange(2, 1, lastRow - 1, lastColumn).getValues();
  const records = values
    .map((row, rowOffset) => {
      return { row: row, rowNumber: rowOffset + 2 };
    })
    .filter((entry) => normalizeText_(entry.row[0]) !== '')
    .map((entry) => {
      const record = { _rowNumber: entry.rowNumber };
      headers.forEach((header, index) => {
        record[header] = entry.row[index];
      });
      return record;
    });
  EXECUTION_RECORD_CACHE_[sheetName] = records;
  return records.map((record) => Object.assign({}, record));
}
/**
 * Appends records inside the worksheet's native Google Sheets Table.
 *
 * The application caches ordered headers and native-table metadata, reserves
 * available rows within the Table, expands the Table when necessary, and
 * writes grouped contiguous ranges with setValues(). SpreadsheetApp.flush()
 * is retained so subsequent operations see the committed append immediately.
 *
 * The Advanced Google Sheets service must remain enabled because native Table
 * metadata and Table resizing use the Sheets API.
 *
 * @param {string} sheetName
 * @param {Object} record
 * @return {number} Worksheet row number containing the appended record.
 */
  function appendRecord_(sheetName, record) {
    return appendRecords_(sheetName, [record])[0];
  }
  function appendRecords_(sheetName, records) {
    if (!Array.isArray(records) || records.length === 0) {
      return [];
    }
    const sheet = getSheetOrThrow_(sheetName);
    const headers = getSheetHeaders_(sheet);
    const primaryHeader = headers[0];
    records.forEach((record) => {
      const primaryValue = Object.prototype.hasOwnProperty.call(
        record,
        primaryHeader
      )
        ? normalizeText_(record[primaryHeader])
        : '';
      if (!primaryValue) {
        throw new Error(
          `Cannot append to ${sheetName}: ${primaryHeader} is required.`
        );
      }
    });
    const appendPerf = sheetName === 'Transactions' ? PT_BUNDLING_APPEND_PERF_ : null;
    let appendPerfAt = appendPerf ? Date.now() : 0;
    if (appendPerf) appendPerf.steps.push({name:'Headers', ms:appendPerfAt-appendPerf.startedAt});
    const table = getNativeTableForSheet_(sheet);
    if (appendPerf) { appendPerf.steps.push({name:'Native table metadata', ms:Date.now()-appendPerfAt}); appendPerfAt=Date.now(); }
    if (!table) {
      throw new Error(
        `No native Google Sheets Table was found on ${sheetName}.`
      );
    }
    const targetRows = reserveEmptyTableRows_(sheet, table, records.length);
    if (appendPerf) { appendPerf.steps.push({name:'Reserve empty row', ms:Date.now()-appendPerfAt}); appendPerfAt=Date.now(); }
    const rowValues = records.map((record) => {
      return headers.map((header) => {
        return Object.prototype.hasOwnProperty.call(record, header)
          ? record[header]
          : '';
      });
    });
    let groupStart = 0;
    while (groupStart < targetRows.length) {
      let groupEnd = groupStart + 1;
      while (
        groupEnd < targetRows.length &&
        targetRows[groupEnd] === targetRows[groupEnd - 1] + 1
      ) {
        groupEnd += 1;
      }
      sheet
        .getRange(
          targetRows[groupStart],
          1,
          groupEnd - groupStart,
          headers.length
        )
        .setValues(rowValues.slice(groupStart, groupEnd));
      groupStart = groupEnd;
    }
    if (appendPerf) { appendPerf.steps.push({name:'setValues', ms:Date.now()-appendPerfAt}); appendPerfAt=Date.now(); }
    SpreadsheetApp.flush();
    if (appendPerf) { appendPerf.steps.push({name:'SpreadsheetApp.flush', ms:Date.now()-appendPerfAt}); appendPerfAt=Date.now(); }
    invalidateRecordCache_(sheetName);
    return targetRows;
  }
  function reserveEmptyTableRows_(sheet, table, count) {
    const firstDataRow = table.range.startRowIndex + 2;
    const lastTableRow = table.range.endRowIndex;
    const availableRows = Math.max(0, lastTableRow - firstDataRow + 1);
    const ids = availableRows
      ? sheet.getRange(firstDataRow, 1, availableRows, 1).getDisplayValues()
      : [];
    const rows = [];
    ids.forEach((row, offset) => {
      if (rows.length < count && normalizeText_(row[0]) === '') {
        rows.push(firstDataRow + offset);
      }
    });
    if (rows.length < count) {
      const expansionRows = Math.max(20, count - rows.length);
      const expandedLastRow = lastTableRow + expansionRows;
      ensureSheetRowCapacity_(sheet, expandedLastRow);
      resizeNativeTable_(table, expandedLastRow);
      for (let row = lastTableRow + 1; rows.length < count; row += 1) {
        rows.push(row);
      }
    }
    return rows;
  }
/**
 * Returns the native Google Sheets Table defined on a worksheet.
 *
 * Inventory data sheets are designed to contain one native Table each.
 *
 * @param {GoogleAppsScript.Spreadsheet.Sheet} sheet
 * @return {Object|null}
 */
function getNativeTableForSheet_(sheet) {
  const sheetId = String(sheet.getSheetId());
  if (!EXECUTION_NATIVE_TABLE_CACHE_) {
    EXECUTION_NATIVE_TABLE_CACHE_ = new Map();
  }
  if (EXECUTION_NATIVE_TABLE_CACHE_.has(sheetId)) {
    const executionTable = EXECUTION_NATIVE_TABLE_CACHE_.get(sheetId);
    return executionTable
      ? JSON.parse(JSON.stringify(executionTable))
      : null;
  }
  const cacheKey = inventoryStructureCacheKey_('NATIVE_TABLE', sheet);
  const cached = readInventoryStructureCache_(cacheKey);
  if (cached && Object.prototype.hasOwnProperty.call(cached, 'table')) {
    EXECUTION_NATIVE_TABLE_CACHE_.set(sheetId, cached.table);
    return cached.table
      ? JSON.parse(JSON.stringify(cached.table))
      : null;
  }
  const spreadsheet = getInventorySpreadsheet_();
  const response = Sheets.Spreadsheets.get(
    spreadsheet.getId(),
    {includeGridData:false}
  );
  (response.sheets || []).forEach((apiSheet) => {
    const apiSheetId = String(apiSheet.properties.sheetId);
    const table = (apiSheet.tables || [])[0] || null;
    EXECUTION_NATIVE_TABLE_CACHE_.set(apiSheetId, table);
    const spreadsheetSheet = spreadsheet.getSheets().find((candidate) => {
      return String(candidate.getSheetId()) === apiSheetId;
    });
    if (spreadsheetSheet) {
      cacheInventoryStructureValue_(
        inventoryStructureCacheKey_('NATIVE_TABLE', spreadsheetSheet),
        {table:table}
      );
    }
  });
  const table = EXECUTION_NATIVE_TABLE_CACHE_.get(sheetId) || null;
  return table ? JSON.parse(JSON.stringify(table)) : null;
}

/**
 * Adds worksheet rows when a native Table needs additional capacity.
 *
 * @param {GoogleAppsScript.Spreadsheet.Sheet} sheet
 * @param {number} requiredRows
 */
function ensureSheetRowCapacity_(
  sheet,
  requiredRows
) {

  const currentRows = sheet.getMaxRows();

  if (currentRows < requiredRows) {

    sheet.insertRowsAfter(
      currentRows,
      requiredRows - currentRows
    );
  }
}

/**
 * Applies sparse row updates through one rectangular read and one write.
 */
function updateRecordRows_(sheetName, rowUpdates) {
  const updatesToApply = Array.isArray(rowUpdates)
    ? rowUpdates.filter((item) => {
        return item && Number(item.rowNumber) >= 2 && item.updates &&
          Object.keys(item.updates).length;
      })
    : [];
  if (!updatesToApply.length) return;
  const sheet = getSheetOrThrow_(sheetName);
  const headerMap = getHeaderMap_(sheet);
  const rowsByNumber = new Map();
  updatesToApply.forEach((item) => {
    const rowNumber = Number(item.rowNumber);
    const merged = rowsByNumber.get(rowNumber) || {};
    Object.entries(item.updates).forEach(([header, value]) => {
      if (!(header in headerMap)) {
        throw new Error(`Unknown ${sheetName} column: ${header}`);
      }
      merged[header] = value;
    });
    rowsByNumber.set(rowNumber, merged);
  });
  const sortedRows = [...rowsByNumber.keys()].sort((a, b) => a - b);
  const groups = [];
  sortedRows.forEach((rowNumber) => {
    const current = groups[groups.length - 1];
    if (!current || rowNumber !== current[current.length - 1] + 1) {
      groups.push([rowNumber]);
    } else {
      current.push(rowNumber);
    }
  });
  groups.forEach((rows) => {
    let minColumn = Infinity;
    let maxColumn = 0;
    rows.forEach((rowNumber) => {
      Object.keys(rowsByNumber.get(rowNumber)).forEach((header) => {
        const column = headerMap[header] + 1;
        minColumn = Math.min(minColumn, column);
        maxColumn = Math.max(maxColumn, column);
      });
    });
    const firstRow = rows[0];
    const range = sheet.getRange(
      firstRow,
      minColumn,
      rows.length,
      maxColumn - minColumn + 1
    );
    const values = range.getValues();
    rows.forEach((rowNumber) => {
      Object.entries(rowsByNumber.get(rowNumber)).forEach(([header, value]) => {
        values[rowNumber - firstRow][headerMap[header] + 1 - minColumn] = value;
      });
    });
    range.setValues(values);
  });
  invalidateRecordCache_(sheetName);
}


/**
 * Reads selected Transaction columns from newest to oldest in bounded chunks.
 * The visitor returns true when enough history has been collected.
 */
function visitTransactionsNewestFirst_(columnNames, visitor, chunkSize = 500) {
  const sheet = getSheetOrThrow_('Transactions');
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;
  const headerMap = getHeaderMap_(sheet);
  const columns = columnNames.map((name) => {
    if (!(name in headerMap)) throw new Error(`Unknown Transactions column: ${name}`);
    return headerMap[name] + 1;
  });
  const firstColumn = Math.min(...columns);
  const lastColumn = Math.max(...columns);
  for (let endRow = lastRow; endRow >= 2;) {
    const startRow = Math.max(2, endRow - chunkSize + 1);
    const values = sheet.getRange(
      startRow,
      firstColumn,
      endRow - startRow + 1,
      lastColumn - firstColumn + 1
    ).getValues();
    for (let index = values.length - 1; index >= 0; index -= 1) {
      const source = values[index];
      const record = {};
      columnNames.forEach((name, columnIndex) => {
        record[name] = source[columns[columnIndex] - firstColumn];
      });
      if (visitor(record, startRow + index) === true) return;
    }
    endRow = startRow - 1;
  }
}

/**
 * Generates a persistent unique application ID.
 *
 * Counts are maintained in Script Properties rather than being
 * calculated from worksheet contents. Deleting an old record therefore
 * cannot cause an identifier to be reused.
 *
 * Examples:
 *
 *   generateId_('PT_GAME', 'PT', 4)
 *     -> PT0001
 *
 *   generateId_('TRANSACTION', 'TX-', 7)
 *     -> TX-0000001
 *
 * @param {string} counterName
 * @param {string} prefix
 * @param {number} digits
 * @return {string}
 */
  function generateId_(counterName, prefix, digits) {
    return generateIds_(counterName, prefix, digits, 1)[0];
  }
  function generateIds_(counterName, prefix, digits, count) {
    const idCount = integerOrThrow_(
      count,
      'ID count',
      { allowZero: false }
    );
    const lock = LockService.getScriptLock();
    lock.waitLock(30000);
    try {
      const properties = PropertiesService.getScriptProperties();
      const propertyKey = `ID_COUNTER_${counterName}`;
      const currentValue = Number(
        properties.getProperty(propertyKey) || 0
      );
      properties.setProperty(
        propertyKey,
        String(currentValue + idCount)
      );
      return Array.from(
        { length: idCount },
        (_, index) => {
          return prefix + String(currentValue + index + 1).padStart(digits, '0');
        }
      );
    } finally {
      lock.releaseLock();
    }
  }

/**
 * Converts a supplied value into a validated integer.
 *
 * @param {*} value
 * @param {string} label
 * @param {Object=} options
 * @return {number}
 */
function integerOrThrow_(
  value,
  label,
  options
) {

  const number =
    Number(value);

  const allowZero =
    !options ||
    options.allowZero !== false;


  const invalidInteger =
    !Number.isInteger(number);

  const invalidPositiveValue =
    !allowZero &&
    number <= 0;

  const invalidNonnegativeValue =
    allowZero &&
    number < 0;


  if (
    invalidInteger ||
    invalidPositiveValue ||
    invalidNonnegativeValue
  ) {

    throw new Error(
      `${label} must be ` +
      `${allowZero ? 'a nonnegative' : 'a positive'} ` +
      'whole number.'
    );
  }


  return number;
}


/**
 * Converts a supplied value into a validated nonnegative number.
 *
 * @param {*} value
 * @param {string} label
 * @return {number}
 */
function numberOrThrow_(
  value,
  label
) {

  const number =
    Number(value);


  if (
    !Number.isFinite(number) ||
    number < 0
  ) {

    throw new Error(
      `${label} must be a nonnegative number.`
    );
  }


  return number;
}

 *
 * @return {Date}

/**

 * Inventory dashboard and consolidated ordering.
 * Scope: Calculates rolling usage, presents materialized inventory health, and records multi-product dashboard orders.
 * Current as of v1.20.
 */
const DASHBOARD_SESSION_WINDOW = 10;
const DASHBOARD_SESSIONS_PER_WEEK = 3;

function getDashboardData() {
  const games = getPTCatalog_().filter(isActiveCatalogRecord_);
  const papers = getPaperCatalog_().filter(isActiveCatalogRecord_);
  const recent = getRecentDashboardTransactions_();
  const recentSessionIds = recent.sessionIds;
  const recentSet = new Set(recentSessionIds);
  const ptSold = new Map();
  const paperSold = new Map();
  recent.transactions.forEach((transaction) => {
    if (
      String(transaction.Transaction_Type) !== 'Sold' ||
      !recentSet.has(String(transaction.Session_ID))
    ) return;
    const key = String(transaction.Product_ID);
    if (String(transaction.Inventory_Type) === 'PT') {
      ptSold.set(key, (ptSold.get(key) || 0) + 1);
    }
    if (String(transaction.Inventory_Type) === 'Paper') {
      paperSold.set(
        key,
        (paperSold.get(key) || 0) - Number(transaction.Quantity_Change || 0)
      );
    }
  });
  const divisor = recentSessionIds.length || 0;
  const pt = games.map((game) => ({
    gameId: String(game.PT_Game_ID),
    formNumber: normalizeFormNumber_(game.Primary_Form_Number),
    label: `${game.Game_Name} (${normalizeFormNumber_(game.Primary_Form_Number)})`,
    onHand: Number(game.Qty_On_Hand || 0),
    targetOnHand: Number(game.Target_On_Hand || 0),
    bundled: Number(game.Qty_Bundled || 0),
    targetBundled: Number(game.Target_Bundled || 0),
    soldPerWeek: divisor
      ? (ptSold.get(String(game.PT_Game_ID)) || 0) / divisor * DASHBOARD_SESSIONS_PER_WEEK
      : 0,
    netProfitPerDollar: Number(game.Net_Income_Per_Patron_Dollar || 0)
  })).sort((a, b) => b.soldPerWeek - a.soldPerWeek || a.label.localeCompare(b.label));
  const paper = papers.map((row) => ({
    paperId: String(row.Paper_ID),
    label: `${row.Short_Name} - ${row.Description}`,
    packsOnHand: Number(row.Qty_On_Hand || 0),
    targetOnHand: Number(row.Target_On_Hand || 0),
    soldPerWeek: divisor
      ? (paperSold.get(String(row.Paper_ID)) || 0) / divisor * DASHBOARD_SESSIONS_PER_WEEK
      : 0,
    netProfitPerDollar: Number(row.Net_Income_Per_Patron_Dollar || 0),
    packsPerCase: Number(row.Packs_Per_Case || 0)
  })).sort((a, b) => b.soldPerWeek - a.soldPerWeek || a.label.localeCompare(b.label));
  return {pt, paper, sessionCount: recentSessionIds.length};
}


function getRecentDashboardTransactions_() {
  const transactions = [];
  const processedSessionIds = [];
  const processedSet = new Set();
  let boundaryFound = false;
  visitTransactionsNewestFirst_(
    [
      'Effective_Date',
      'Inventory_Type',
      'Product_ID',
      'Transaction_Type',
      'Quantity_Change',
      'Session_ID'
    ],
    (transaction) => {
      const isProcessedMarker =
        String(transaction.Inventory_Type) === 'System' &&
        String(transaction.Transaction_Type) === 'Session Processed' &&
        normalizeText_(transaction.Session_ID);
      if (isProcessedMarker) {
        const sessionId = String(transaction.Session_ID);
        if (!processedSet.has(sessionId)) {
          if (processedSessionIds.length >= DASHBOARD_SESSION_WINDOW) {
            boundaryFound = true;
            return true;
          }
          processedSet.add(sessionId);
          processedSessionIds.push(sessionId);
        }
      }
      transactions.push(transaction);
      return false;
    }
  );
  return {
    transactions: transactions,
    sessionIds: processedSessionIds,
    boundaryFound: boundaryFound
  };
}

function isActiveCatalogRecord_(record) {
  return record.Active === true || String(record.Active).toLowerCase() === 'true';
}

function placeDashboardOrder(adminCode, order) {

  if (!authentication.ok) throw new Error(authentication.message);
  try {
    return placeDashboardOrderWithToken_(authentication.token, order);

  }
}

function placeDashboardOrderWithToken_(token, order) {
  requireAdmin_(token);
  order = order || {};
  const ptLines = Array.isArray(order.pt) ? order.pt : [];
  const paperLines = Array.isArray(order.paper) ? order.paper : [];
  if (!ptLines.length && !paperLines.length) {
    throw new Error('Enter at least one order quantity.');
  }
  const games = getPTCatalog_();
  const papers = getPaperCatalog_();
  const gamesById = new Map(games.map((row) => [String(row.PT_Game_ID), row]));
  const papersById = new Map(papers.map((row) => [String(row.Paper_ID), row]));
  const validatedPT = ptLines.map((line) => {
    const quantity = integerOrThrow_(line.quantity, 'PT deals to order', {allowZero:false});
    const game = gamesById.get(String(line.gameId));
    if (!game || !isActiveCatalogRecord_(game)) {
      throw new Error('An ordered PT game is unavailable.');
    }
    return {game: game, quantity: quantity};
  });
  const validatedPaper = paperLines.map((line) => {
    const quantity = integerOrThrow_(line.quantity, 'Paper cases to order', {allowZero:false});
    const paper = papersById.get(String(line.paperId));
    if (!paper || !isActiveCatalogRecord_(paper)) {
      throw new Error('An ordered Paper type is unavailable.');
    }
    return {
      paper: paper,
      quantity: quantity,
      packsPerCase: integerOrThrow_(paper.Packs_Per_Case, 'Packs per Case', {allowZero:false})
    };
  });
  const lock = LockService.getDocumentLock();
  lock.waitLock(30000);
  try {
    const effectiveDate = now_();
    const ptRecords = [];
    const paperRecords = [];
    const transactionEntries = [];
    const ptDeltas = new Map();
    const paperDeltas = new Map();
    validatedPT.forEach(({game, quantity}) => {
      const gameId = String(game.PT_Game_ID);
      const formNumber = normalizeFormNumber_(game.Primary_Form_Number);
      const dealIds = generateIds_('PT_DEAL', 'PTD-', 6, quantity);
      dealIds.forEach((dealId) => {
        ptRecords.push({
          PT_Deal_ID: dealId, PT_Game_ID: gameId, Form_Number: formNumber,
          Serial_Number: '', Received_Date: effectiveDate, Deal_Status: 'Unbundled',
          Bundled_Date: '', Expended_Date: '', Expended_Session_ID: '',
          Notes: `Dashboard order: ${quantity} deal(s).`
        });
        transactionEntries.push({
          effectiveDate: effectiveDate, inventoryType: 'PT', productId: gameId,
          itemId: dealId, formNumber: formNumber, transactionType: 'Received',
          quantityChange: 1, unit: 'Deal', source: 'Inventory Ordering',
          notes: `Dashboard order: ${quantity} deal(s).`
        });
      });
      const delta = ptDeltas.get(gameId) || {unbundled:0,bundled:0,open:0,onHand:0};
      delta.unbundled += quantity;
      delta.onHand += quantity;
      ptDeltas.set(gameId, delta);
    });
    validatedPaper.forEach(({paper, quantity, packsPerCase}) => {
      const paperId = String(paper.Paper_ID);
      const caseIds = generateIds_('PAPER_CASE', 'PC-', 6, quantity);
      caseIds.forEach((caseId) => {
        paperRecords.push({
          Paper_Case_ID: caseId, Paper_ID: paperId, Serial_Number: '',
          Received_Date: effectiveDate, Case_Status: 'Unopened', Opened_Date: '',
          Expended_Date: '', Notes: `Dashboard order: ${quantity} case(s).`,
          Current_Packs: packsPerCase, Balance_Updated_At: effectiveDate
        });
        transactionEntries.push({
          effectiveDate: effectiveDate, inventoryType: 'Paper', productId: paperId,
          itemId: caseId, transactionType: 'Received', quantityChange: packsPerCase,
          unit: 'Pack', source: 'Inventory Ordering',
          notes: `Dashboard order: ${quantity} case(s).`
        });
      });
      paperDeltas.set(
        paperId,
        (paperDeltas.get(paperId) || 0) + quantity * packsPerCase
      );
    });
    if (ptRecords.length) appendRecords_('PT_Deals', ptRecords);
    if (paperRecords.length) appendRecords_('Paper_Cases', paperRecords);
    try {
      postInventoryTransactions_(transactionEntries, {updatePaperBalances:false});
    } catch (error) {
      if (ptRecords.length) {
        deleteRowsByIdSet_('PT_Deals','PT_Deal_ID',new Set(ptRecords.map((row)=>row.PT_Deal_ID)));
      }
      if (paperRecords.length) {
        deleteRowsByIdSet_('Paper_Cases','Paper_Case_ID',new Set(paperRecords.map((row)=>row.Paper_Case_ID)));
      }
      throw new Error(`Dashboard order failed and was rolled back: ${error.message}`);
    }
    applyPTCatalogBalanceDeltas_(ptDeltas);
    applyPaperCatalogBalanceDeltas_(paperDeltas);
    return {
      ptDeals: ptRecords.length,
      paperCases: paperRecords.length,
      lineCount: validatedPT.length + validatedPaper.length
    };
  } finally {
    lock.releaseLock();
  }
}

/**
 * Finds an identified PT deal by authoritative PT_Game_ID + Serial_Number.
 *
 * Blank serial numbers deliberately do not identify a deal because multiple
 * anonymous deals of one PT game are valid inventory records.
 */
function findPTDeal_(gameId, serialNumber) {
  const normalizedGameId = normalizeText_(gameId);
  const normalizedSerialNumber = normalizeText_(serialNumber);
  if (!normalizedGameId || !normalizedSerialNumber) return null;
  return getPTDeals_().find((deal) => {
    return String(deal.PT_Game_ID) === normalizedGameId &&
      normalizeText_(deal.Serial_Number) === normalizedSerialNumber;
  }) || null;
}


/**
 * Creates one physical PT deal record.
 *
 * Serial_Number may be blank while the deal is unidentified. Duplicate
 * protection is enforced only after a nonblank serial number exists.
 */
function createPTDeal_(data) {
  if (!PT_DEAL_STATUSES.includes(data.status)) {
    throw new Error(`Invalid PT Deal Status: ${data.status}`);
  }
  const gameId = normalizeText_(data.gameId);
  const primaryFormNumber = getPrimaryPTFormNumber_(gameId, {includeInactive:true});
  const normalizedSerialNumber = normalizeText_(data.serialNumber);
  if (normalizedSerialNumber && findPTDeal_(gameId, normalizedSerialNumber)) {
    throw new Error(
      `Deal ${primaryFormNumber}/${normalizedSerialNumber} already exists.`
    );
  }
  const dealId = generateId_('PT_DEAL', 'PTD-', 6);
  const receivedDate = data.receivedDate || now_();
  appendRecord_('PT_Deals', {
    PT_Deal_ID: dealId,
    PT_Game_ID: gameId,
    Form_Number: primaryFormNumber,
    Serial_Number: normalizedSerialNumber,
    Received_Date: receivedDate,
    Deal_Status: data.status,
    Bundled_Date: data.status === 'Bundled' ? receivedDate : '',
    Expended_Date: ['Sold','Destroyed','Returned'].includes(data.status)
      ? receivedDate : '',
    Expended_Session_ID: data.sessionId || '',
    Notes: data.notes || ''
  });
  return dealId;
}


function normalizePTFormNumbersV119c(token) {
  requireAdmin_(token);
  const lock = LockService.getDocumentLock();
  lock.waitLock(30000);
  try {
    const primaryByGameId = new Map(
      getPTCatalog_().map((game) => [
        String(game.PT_Game_ID),
        normalizeFormNumber_(game.Primary_Form_Number)
      ])
    );
    const specifications = [
      {sheetName:'PT_Deals', productField:'PT_Game_ID'},
      {sheetName:'Transactions', productField:'Product_ID', ptOnly:true},
      {sheetName:'Inventory_Issues', productField:'Product_ID', ptOnly:true},
      {sheetName:'Reconciliation_Records', productField:'Product_ID', ptOnly:true}
    ];
    const changed = {};
    specifications.forEach((specification) => {
      const updates = [];
      getRecords_(specification.sheetName).forEach((record) => {
        if (specification.ptOnly && String(record.Inventory_Type) !== 'PT') return;
        const primary = primaryByGameId.get(String(record[specification.productField]));
        if (!primary || normalizeFormNumber_(record.Form_Number) === primary) return;
        updates.push({
          rowNumber: record._rowNumber,
          updates: {Form_Number: primary}
        });
      });
      updateRecordRows_(specification.sheetName, updates);
      changed[specification.sheetName] = updates.length;
    });
    return {ok:true, version:'1.19c', changed:changed};
  } finally {
    lock.releaseLock();
  }
}

function buildInventoryTransactionRecord_(data, transactionId, timestamp) {
  const quantityChange = Number(data.quantityChange);
  if (!Number.isInteger(quantityChange)) {
    throw new Error('Transaction Quantity_Change must be an integer.');
  }
  return {
    Transaction_ID: transactionId,

    Item_ID: data.itemId || '',
    Form_Number: data.formNumber || '',
    Serial_Number: data.serialNumber || '',
    Transaction_Type: data.transactionType,
    Quantity_Change: quantityChange,

    Reconciliation_ID: data.reconciliationId || '',
    Notes: data.notes || ''
  };
}

function postInventoryTransactions_(entries, options = {}) {
  if (!entries.length) return [];
  const timestamp = now_();
  const ids = generateIds_('TRANSACTION', 'TX-', 7, entries.length);
  const records = entries.map((entry, index) => {
    return buildInventoryTransactionRecord_(entry, ids[index], timestamp);
  });
  appendRecords_('Transactions', records);

  const paperDeltas = new Map();
  entries.forEach((entry) => {
    if (String(entry.inventoryType) === 'Paper' && entry.itemId) {
      const key = String(entry.itemId);
      paperDeltas.set(key, (paperDeltas.get(key) || 0) + Number(entry.quantityChange));
    }
  });
  if (paperDeltas.size && options.updatePaperBalances !== false) {
    applyPaperBalanceDeltas_(paperDeltas, options.refreshPaperCatalog !== false);
  }
  return ids;
}

/**

 *
 * This is important for a multi-deal order. All product-level conditions are
 * checked once before the first deal or transaction is created.
 */
function validateAnonymousPTStockAddition_(data) {
  const quantity = integerOrThrow_(data.quantity, 'PT quantity', {allowZero:false});
  const gameId = normalizeText_(data.gameId);
  if (!gameId) throw new Error('PT Game ID is required.');
  const game = getPTGameById_(gameId);
  if (!game) throw new Error(`PT Game ${gameId} was not found in PT_Catalog.`);
  if (data.status && data.status !== 'Unbundled') {
    throw new Error('Anonymous ordered PT stock must enter inventory as Unbundled.');
  }
  return {
    quantity: quantity,
    gameId: gameId,
    formNumber: normalizeFormNumber_(game.Primary_Form_Number)
  };
}

/**
 * Best-effort rollback used only for records created during the current failed
 * multi-deal operation.
 *
 * This does not rewrite historical inventory. It exists to prevent an order
 * from being partially posted if an unexpected application/API failure occurs
 * after pre-validation succeeds.
 */
/**
 * Deletes rows matching IDs created by the currently failing operation.
 * Rows are deleted from bottom to top to avoid row-number shifting.
 */
function deleteRowsByIdSet_(
  sheetName,
  idColumnName,
  ids
) {
  if (!ids || ids.size === 0) {
    return;
  }

  const sheet = getSheetOrThrow_(sheetName);
  const records = getRecords_(sheetName);

  const rowsToDelete = records
    .filter((record) => {
      return ids.has(
        String(record[idColumnName])
      );
    })
    .map((record) => record._rowNumber)
    .sort((firstRow, secondRow) => {
      return secondRow - firstRow;
    });

  rowsToDelete.forEach((rowNumber) => {
    sheet.deleteRow(rowNumber);
  });
    invalidateRecordCache_(sheetName);
}

/**
 * Builds the current pack balance for every Paper case from the
 * authoritative Transactions ledger.
 *
 * This function is used only for initialization, validation, and
 * maintenance operations. Normal inventory operations use the
 * materialized Current_Packs values instead of rescanning Transactions.
 *
 * @param {Object[]} transactions
 * @return {Map<string, number>}
 */
function buildPaperBalanceMap_(transactions) {

  const balancesByCaseId =
    new Map();


  (transactions || []).forEach(
    (transaction) => {

      if (
        String(transaction.Inventory_Type) !==
        'Paper'
      ) {

        return;
      }


      const paperCaseId =
        normalizeText_(
          transaction.Item_ID
        );


      if (!paperCaseId) {

        return;
      }


      const quantityChange =
        Number(
          transaction.Quantity_Change || 0
        );


      if (
        !Number.isFinite(
          quantityChange
        )
      ) {

        throw new Error(
          `Transaction ${transaction.Transaction_ID || '(unknown)'} ` +
          'contains an invalid Paper quantity.'
        );
      }


      const currentBalance =
        balancesByCaseId.get(
          paperCaseId
        ) || 0;


      balancesByCaseId.set(
        paperCaseId,
        currentBalance + quantityChange
      );
    }
  );


  return balancesByCaseId;
}

function ptStatusContribution_(status) {
  return {
    unbundled: status === 'Unbundled' ? 1 : 0,
    bundled: status === 'Bundled' ? 1 : 0,
    open: status === 'Open' ? 1 : 0,
    onHand: ['Unbundled', 'Bundled'].includes(String(status)) ? 1 : 0
  };
}

function applyPTCatalogBalanceDeltas_(deltasByGameId) {
  if (!deltasByGameId || !deltasByGameId.size) return;
  const updatedAt = now_();
  const updates = getPTCatalog_().filter((row) => {
    return deltasByGameId.has(String(row.PT_Game_ID));
  }).map((row) => {
    const delta = deltasByGameId.get(String(row.PT_Game_ID));
    return {
      rowNumber: row._rowNumber,
      updates: {
        Qty_Unbundled: Number(row.Qty_Unbundled || 0) + delta.unbundled,
        Qty_Bundled: Number(row.Qty_Bundled || 0) + delta.bundled,
        Qty_Open: Number(row.Qty_Open || 0) + delta.open,
        Qty_On_Hand: Number(row.Qty_On_Hand || 0) + delta.onHand,
        Balance_Updated_At: updatedAt
      }
    };
  });
  updateRecordRows_('PT_Catalog', updates);
}

function recordPTStatusChange_(deltasByGameId, gameId, oldStatus, newStatus) {
  if (oldStatus === newStatus) return;
  const before = ptStatusContribution_(oldStatus);
  const after = ptStatusContribution_(newStatus);
  const key = String(gameId);
  const delta = deltasByGameId.get(key) || {unbundled: 0, bundled: 0, open: 0, onHand: 0};
  Object.keys(delta).forEach((name) => delta[name] += after[name] - before[name]);
  deltasByGameId.set(key, delta);
}

  function applySinglePTStatusChange_(gameId, oldStatus, newStatus) {

    recordPTStatusChange_(
      deltasByGameId,
      gameId,
      String(oldStatus || ''),
      String(newStatus || '')
    );
    applyPTCatalogBalanceDeltas_(deltasByGameId);
  }
  function refreshPTCatalogBalances_() {
  const counts = new Map();
  getPTDeals_().forEach((deal) => {
    const id = String(deal.PT_Game_ID);
    const value = counts.get(id) || {unbundled: 0, bundled: 0, open: 0};
    if (deal.Deal_Status === 'Unbundled') value.unbundled += 1;
    if (deal.Deal_Status === 'Bundled') value.bundled += 1;
    if (deal.Deal_Status === 'Open') value.open += 1;
    counts.set(id, value);
  });
  const updatedAt = now_();
  updateRecordRows_('PT_Catalog', getPTCatalog_().map((row) => {
    const value = counts.get(String(row.PT_Game_ID)) || {unbundled: 0, bundled: 0, open: 0};
    return {rowNumber: row._rowNumber, updates: {
      Qty_Unbundled: value.unbundled,
      Qty_Bundled: value.bundled,
      Qty_Open: value.open,
      Qty_On_Hand: value.unbundled + value.bundled,
      Balance_Updated_At: updatedAt
    }};
  }));
}

function applyPaperCatalogBalanceDeltas_(deltasByPaperId) {

}

  getPaperCases_().forEach((row) => {

function applyPaperBalanceDeltas_(deltasByCaseId, refreshCatalog = true) {
  if (!deltasByCaseId || !deltasByCaseId.size) return;
  const updatedAt = now_();
  const paperDeltas = new Map();
  const updates = getPaperCases_().filter((row) => {
    return deltasByCaseId.has(String(row.Paper_Case_ID));
  }).map((row) => {
    const delta = Number(deltasByCaseId.get(String(row.Paper_Case_ID)) || 0);
    const paperId = String(row.Paper_ID);
    paperDeltas.set(paperId, (paperDeltas.get(paperId) || 0) + delta);
    return {
      rowNumber: row._rowNumber,
      updates: {
        Current_Packs: Number(row.Current_Packs || 0) + delta,
        Balance_Updated_At: updatedAt
      }
    };
  });
  if (updates.length !== deltasByCaseId.size) {

function validateMaterializedBalances_() {
  const errors = [];
  const ledger = buildPaperBalanceMap_(getRecords_('Transactions'));
  getPaperCases_().forEach((paperCase) => {
    const expected = ledger.get(String(paperCase.Paper_Case_ID)) || 0;
    const actual = Number(paperCase.Current_Packs || 0);
    if (expected !== actual) errors.push(`Paper case ${paperCase.Paper_Case_ID}: ledger ${expected}, current ${actual}.`);
  });
  const ptCounts = new Map();
  getPTDeals_().forEach((deal) => {
    const key = String(deal.PT_Game_ID);
    const c = ptCounts.get(key) || {unbundled: 0, bundled: 0, open: 0, onHand: 0};
    const v = ptStatusContribution_(String(deal.Deal_Status));
    Object.keys(c).forEach((name) => c[name] += v[name]);
    ptCounts.set(key, c);
  });
  getPTCatalog_().forEach((game) => {
    const c = ptCounts.get(String(game.PT_Game_ID)) || {unbundled: 0, bundled: 0, open: 0, onHand: 0};
    [['Qty_Unbundled','unbundled'],['Qty_Bundled','bundled'],['Qty_Open','open'],['Qty_On_Hand','onHand']].forEach(([field,key]) => {
      if (Number(game[field] || 0) !== c[key]) errors.push(`PT game ${game.PT_Game_ID}.${field}: expected ${c[key]}, found ${game[field] || 0}.`);
    });
  });
  return {valid: errors.length === 0, errors: errors};
}

function serializeInventoryIssueDate_(value) {
  if (!value) return '';
  if (value instanceof Date) {
    return isNaN(value.getTime()) ? '' : value.toISOString();
  }
  const parsed = new Date(value);
  return isNaN(parsed.getTime()) ? String(value) : parsed.toISOString();
}

function getInventoryIssues(token, statusFilter) {
  requireAdmin_(token);
  const normalizedFilter = normalizeText_(statusFilter) || 'Open';
  const validFilters = ['Open', 'Resolved', 'All'];
  if (!validFilters.includes(normalizedFilter)) {
    throw new Error('Invalid Inventory Issue filter.');
  }
  const ptById = new Map(
    getPTCatalog_().map((game) => [String(game.PT_Game_ID), game])
  );
  const paperById = new Map(
    getPaperCatalog_().map((paper) => [String(paper.Paper_ID), paper])
  );
  return getRecords_('Inventory_Issues')
    .filter((issue) => {
      return normalizedFilter === 'All' ||
        String(issue.Status) === normalizedFilter;
    })
    .map((issue) => {
      const inventoryType = String(issue.Inventory_Type);
      const productId = String(issue.Product_ID || '');
      const game = inventoryType === 'PT'
        ? ptById.get(productId)
        : null;
      const paper = inventoryType === 'Paper'
        ? paperById.get(productId)
        : null;
      let productLabel = productId || '(not identified)';
      if (game) {
        productLabel = String(game.Game_Name);
      } else if (inventoryType === 'PT' && issue.Form_Number) {
        productLabel = String(issue.Form_Number);
      } else if (paper) {
        productLabel = `${paper.Short_Name} - ${paper.Description}`;
      }
      return {
        issueId: String(issue.Issue_ID),
        timestamp: serializeInventoryIssueDate_(issue.Timestamp),
        inventoryType: inventoryType,
        issueType: String(issue.Issue_Type),
        productId: productId,
        productLabel: productLabel,
        itemId: String(issue.Item_ID || ''),
        formNumber: game
          ? normalizeFormNumber_(game.Primary_Form_Number)
          : String(issue.Form_Number || ''),
        serialNumber: String(issue.Serial_Number || ''),
        sessionId: String(issue.Session_ID || ''),
        expectedQuantity: Number(issue.Expected_Quantity || 0),
        reportedQuantity: Number(issue.Reported_Quantity || 0),
        discrepancyAmount: Number(issue.Discrepancy_Amount || 0),
        status: String(issue.Status),
        reconciliationId: String(issue.Reconciliation_ID || ''),
        notes: String(issue.Notes || ''),
        resolvedAt: serializeInventoryIssueDate_(issue.Resolved_At),
        resolutionNotes: String(issue.Resolution_Notes || '')
      };
    })
    .sort((first, second) => {
      const firstOpen = first.status === 'Open' ? 0 : 1;
      const secondOpen = second.status === 'Open' ? 0 : 1;
      if (firstOpen !== secondOpen) return firstOpen - secondOpen;
      return new Date(second.timestamp).getTime() -
        new Date(first.timestamp).getTime();
    });
}


function resolveInventoryIssue(token, issueId, resolutionNotes) {
  requireAdmin_(token);

  const normalizedIssueId = normalizeText_(issueId);
  const normalizedNotes = normalizeText_(resolutionNotes);

  if (!normalizedIssueId) {
    throw new Error('Inventory Issue ID is required.');
  }

  if (!normalizedNotes) {
    throw new Error('Resolution Notes are required.');
  }

  const issue = getRecords_('Inventory_Issues').find((candidate) => {
    return String(candidate.Issue_ID) === normalizedIssueId;
  });

  if (!issue) {
    throw new Error(`Inventory Issue ${normalizedIssueId} was not found.`);
  }

  if (String(issue.Status) !== 'Open') {
    throw new Error(
      `Inventory Issue ${normalizedIssueId} is already ${issue.Status}.`
    );
  }

  updateRecordRow_(
    'Inventory_Issues',
    issue._rowNumber,
    {
      Status: 'Resolved',
      Resolved_At: now_(),
      Resolution_Notes: normalizedNotes
    }
  );

  return {
    issueId: normalizedIssueId,
    status: 'Resolved'
  };
}

 * Scope: Validates catalog selections and adds anonymous PT deals or Paper cases with receipt Transactions.

function recordPTOrder(token, payload) {
  requireAdmin_(token);
  payload = payload || {};
  const gameId = normalizeText_(payload.gameId);
  const catalogGame = getPTGameById_(gameId);
  if (!catalogGame) {
    throw new Error(`PT Game ${gameId} was not found in PT_Catalog.`);
  }
  const primaryFormNumber = normalizeFormNumber_(catalogGame.Primary_Form_Number);
  const catalogDealsPerCase = integerOrThrow_(
    catalogGame.Deals_Per_Case,
    'Catalog Deals per Case',
    {allowZero:false}
  );
  const casesOrdered = integerOrThrow_(payload.cases, 'Cases Ordered');
  const looseDeals = integerOrThrow_(payload.looseDeals || 0, 'Loose Deals');
  const quantity = casesOrdered * catalogDealsPerCase + looseDeals;
  if (quantity <= 0) throw new Error('The order must add at least one PT deal.');
  const effectiveDate = payload.effectiveDate
    ? new Date(`${payload.effectiveDate}T12:00:00`) : now_();
  const notes = normalizeText_(payload.notes) ||
    `${casesOrdered} case(s) and ${looseDeals} loose deal(s) ordered.`;
  const lock = LockService.getDocumentLock();
  lock.waitLock(30000);
  try {
    const dealIds = createAnonymousPTStock_({
      gameId: gameId,
      formNumber: primaryFormNumber,
      quantity: quantity,
      status: 'Unbundled',
      effectiveDate: effectiveDate,
      transactionType: 'Received',
      source: 'Inventory Ordering',
      notes: notes
    });
    return {
      quantity: quantity,
      casesOrdered: casesOrdered,
      looseDeals: looseDeals,
      dealsPerCase: catalogDealsPerCase,
      dealIds: dealIds,
      formNumber: primaryFormNumber
    };
  } finally {
    lock.releaseLock();
  }
}


function recordPaperOrder(token, payload) {
  requireAdmin_(token);

  payload = payload || {};

  const casesOrdered = integerOrThrow_(

const PT_BUNDLING_PERF_VERSION = '1.19c';
function createPTBundlingTimer_() {
  const startedAt = Date.now();
  let checkpointAt = startedAt;
  const steps = [];
  return {
    mark: function(name, details) {
      const now = Date.now();
      steps.push({name:name, ms:now-checkpointAt, elapsedMs:now-startedAt, details:details || ''});
      checkpointAt = now;
    },
    finish: function(details) {
      return {version:PT_BUNDLING_PERF_VERSION, totalMs:Date.now()-startedAt, details:details || '', steps:steps};
    }
  };
}

function getPTDealsForBundlingLookup_() {
  const sheet = getSheetOrThrow_('PT_Deals');
  const headerMap = getHeaderMap_(sheet);
  const requiredHeaders = [
    'PT_Deal_ID',
    'PT_Game_ID',
    'Form_Number',
    'Serial_Number',
    'Deal_Status',
    'Notes'
  ];
  requiredHeaders.forEach((header) => {
    if (!(header in headerMap)) {
      throw new Error(`Missing PT_Deals column: ${header}`);
    }
  });
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  const columns = requiredHeaders.map((header) => headerMap[header] + 1);
  const firstColumn = Math.min(...columns);
  const lastColumn = Math.max(...columns);
  const values = sheet.getRange(
    2,
    firstColumn,
    lastRow - 1,
    lastColumn - firstColumn + 1
  ).getValues();
  return values.map((row, index) => {
    const record = {_rowNumber:index + 2};
    requiredHeaders.forEach((header) => {
      record[header] = row[headerMap[header] + 1 - firstColumn];
    });
    return record;
  }).filter((record) => normalizeText_(record.PT_Deal_ID));
}

function findPTDealForBundling_(deals, gameId, serialNumber) {

  const normalizedSerial = normalizeText_(serialNumber);

    return String(deal.PT_Game_ID) === normalizedGameId &&

}
function bundlePTDeal(formNumber, serialNumber) {
  const timer = createPTBundlingTimer_();
  const normalizedForm = normalizeFormNumber_(formNumber);
  const normalizedSerial = normalizeText_(serialNumber);
  timer.mark('Normalize input');
  if (!normalizedForm || !normalizedSerial) throw new Error('Enter both a Form Number and Serial Number.');
  const resolvedForm = resolvePTGameByForm_(normalizedForm);
  timer.mark('Catalog / form lookup');
  if (!resolvedForm) throw new Error(`Form Number ${normalizedForm} was not found.`);
  const game = resolvedForm.game;
  const gameId = resolvedForm.gameId;
  const primaryFormNumber = resolvedForm.primaryFormNumber;
  timer.mark('Game lookup');
  if (!game) throw new Error(`Game ${gameId} was not found.`);
  const lock = LockService.getDocumentLock();
  const lockAt = Date.now();
  lock.waitLock(30000);
  timer.mark('Document lock', `waitMs=${Date.now()-lockAt}`);
  try {
    const bundlingDeals = getPTDealsForBundlingLookup_();
    const existingDeal = findPTDealForBundling_(
      bundlingDeals,
      gameId,
      normalizedSerial
    );
    timer.mark('PT Deals lookup');
    let result;
    if (existingDeal) {
      result = bundleExistingIdentifiedPTDeal_({deal:existingDeal,game:game,gameId:gameId,formNumber:primaryFormNumber,serialNumber:normalizedSerial});
    } else {
      result = bundleNewlyIdentifiedPTDeal_({
        game:game,
        gameId:gameId,
        formNumber:primaryFormNumber,
        serialNumber:normalizedSerial,
        bundlingDeals:bundlingDeals
      });
    }
    timer.mark('Inventory writes + transaction');
    const performance = timer.finish(`game=${gameId}; enteredForm=${normalizedForm}; action=${result.action}`);
    performance.append = result.appendPerformance || null;
    delete result.appendPerformance;
    return Object.assign(result, {
      gameId:gameId,gameName:String(game.Game_Name),formNumber:primaryFormNumber,
      serialNumber:normalizedSerial,cards:String(game.Cards_Per_Deal || ''),payout:String(game.Payout_Per_Deal || ''),
      performance:performance
    });
  } finally { lock.releaseLock(); }
}


function bundleExistingIdentifiedPTDeal_(context) {
  const deal = context.deal;
  const status = String(deal.Deal_Status);

  if (status === 'Bundled') {
    throw new Error(
      `Deal ${context.formNumber}/${context.serialNumber} is already bundled.`
    );
  }

  if (status === 'Unbundled') {
    updateRecordRow_('PT_Deals', deal._rowNumber, {
      Deal_Status: 'Bundled',
      Bundled_Date: now_(),
      Notes: appendInventoryNote_(
        deal.Notes,
        'Bundled through integrated PT Processing.'
      )
      });
      applySinglePTStatusChange_(
        deal.PT_Game_ID,
        status,
        'Bundled'
      );
      const bundledTransactionId = postPTBundledTransaction_(
        deal.PT_Deal_ID,
      context.game.PT_Game_ID,
      context.formNumber,
      context.serialNumber,
      'Known identified Unbundled deal was bundled.'
    );

    return {
      dealId: String(deal.PT_Deal_ID),
      action: 'Bundled existing identified deal',
        appendPerformance: consumePTBundlingAppendPerformance_(),
      adjustmentTransactionId: '',
      bundledTransactionId: bundledTransactionId,
      issueId: '',
      resolvedIssueIds: []
    };
  }

  if (status === 'Missing') {
    updateRecordRow_('PT_Deals', deal._rowNumber, {
      Deal_Status: 'Bundled',
      Bundled_Date: now_(),
      Notes: appendInventoryNote_(
        deal.Notes,
        'Missing deal recovered and bundled through integrated PT Processing.'
      )
      });
      applySinglePTStatusChange_(
        deal.PT_Game_ID,
        status,
        'Bundled'
      );
    const adjustmentTransactionId = postInventoryTransaction_({
      effectiveDate: now_(),
      inventoryType: 'PT',
      productId: context.game.PT_Game_ID,
      itemId: deal.PT_Deal_ID,
      formNumber: context.formNumber,
      serialNumber: context.serialNumber,
      transactionType: 'Adjustment',
      quantityChange: 1,
      unit: 'Deal',
      source: 'PT Processing',
      notes: 'Missing deal recovered during PT bundling.'
    });

    const resolvedIssueIds = resolveOpenPTIssuesForDeal_(
      deal,
      'Resolved when the missing deal was recovered during PT bundling.'
    );

    const bundledTransactionId = postPTBundledTransaction_(
      deal.PT_Deal_ID,
      context.game.PT_Game_ID,
      context.formNumber,
      context.serialNumber,
      'Recovered Missing deal was bundled.'
    );

    return {
      dealId: String(deal.PT_Deal_ID),
      action: 'Recovered missing deal and bundled it',
      adjustmentTransactionId: adjustmentTransactionId,
      bundledTransactionId: bundledTransactionId,
      issueId: '',
      resolvedIssueIds: resolvedIssueIds
    };
  }

  throw new Error(
    `Deal ${context.formNumber}/${context.serialNumber} has status ${status} ` +
    'and cannot be bundled. Use reconciliation or administrative review.'
  );
}

function postPTBundledTransaction_(dealId, gameId, formNumber, serialNumber, notes) {

  const idAt = Date.now();

  perf.steps.push({name:'Transaction ID', ms:Date.now()-idAt});

function resolveOpenPTIssuesForDeal_(deal, resolutionNote) {
  const resolvedIssueIds = [];

  getRecords_('Inventory_Issues')
    .filter((issue) => {
      const sameItem =
        String(issue.Item_ID) === String(deal.PT_Deal_ID);

      const sameSerial =
          String(issue.Product_ID) === String(deal.PT_Game_ID) &&
          normalizeText_(issue.Serial_Number) ===
            normalizeText_(deal.Serial_Number);

      return (
        String(issue.Inventory_Type) === 'PT' &&
        String(issue.Status) === 'Open' &&
        (sameItem || sameSerial)
      );
    })
    .forEach((issue) => {
        resolveInventoryIssueRecord_(issue, resolutionNote);

      resolvedIssueIds.push(String(issue.Issue_ID));
    });

  return resolvedIssueIds;
}

const PAPER_CASE_STATUSES = [

];


/**
 * Derives the required physical status from a reconciled pack count.
 * Missing remains an explicit administrator choice. An already-opened case
 * does not become Unopened merely because a correction restores full quantity.
 */
function derivePaperCaseStatus_(
  requestedStatus,
  recordedStatus,
  observedPacks,
  packsPerCase
) {
  if (requestedStatus === 'Missing') {
    return 'Missing';
  }
  if (observedPacks === 0) {
    return 'Expended';
  }
  if (observedPacks < packsPerCase) {
    return 'Opened';
  }
  if (recordedStatus === 'Opened') {
    return 'Opened';
  }
  return requestedStatus;
}

      return second.folderName.localeCompare(first.folderName);
    });

    output[label] = rows;
  });

  return output;
}

function dailyPaperworkMaintenance() {
  validatePaperworkConfiguration_();

  const lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    const today = paperworkToday_();
    const archiveResults = archivePastSessionPaperwork_(today);
    const registry = getPaperworkSessionsRegistry_();

    const todayRecords = getPaperworkSessionRecords_(registry)
      .filter((record) => {
        return (
          record.date &&
          formatPaperworkIsoDate_(record.date) ===
            formatPaperworkIsoDate_(today)
        );
      });

    if (todayRecords.length === 0) {
      return {
        archiveResults: archiveResults,
        creationResult: null,
        message:
          'Past-session processing completed. No session is listed for today.'
      };
    }

    if (todayRecords.length > 1) {
      throw new Error(
        'More than one Sessions row is listed for today. ' +
        'No paperwork was created.'
      );
    }

    const record = todayRecords[0];
    validatePaperworkLabel_(record.label);

    const creationResult = createOnePaperworkSet_(
      today,
      record.label,
      registry
    );

    return {
      archiveResults: archiveResults,
      creationResult: creationResult,
      message: creationResult.message
    };
  } finally {
    lock.releaseLock();
  }
}

  } finally {

function createOnePaperworkSet_(date, label, registry) {
  validatePaperworkLabel_(label);

  const dateCode = formatPaperworkMMDDYY_(date);
  const weeklyFolder = DriveApp.getFolderById(
    PAPERWORK_CONFIG.weeklyFolders[label]
  );

  const existingFolders = collectPaperworkIterator_(
    weeklyFolder.getFoldersByName(dateCode)
  );

  if (existingFolders.length > 1) {
    throw new Error(
      `Multiple weekly folders named ${dateCode} already exist.`
    );
  }

  if (existingFolders.length === 1) {
    const folder = existingFolders[0];
    const expectedSheet = findUniquePaperworkSheet_(
      folder,
      `LBA-Bingo_${dateCode}`
    );

    const expectedPdf = collectPaperworkIterator_(
      folder.getFilesByName(`LBA-fw2g_${dateCode}.pdf`)
    );

    if (expectedSheet && expectedPdf.length === 1) {
      return {
        ok: true,
        created: false,
        dateString: formatPaperworkIsoDate_(date),
        displayDate: formatPaperworkDisplayDate_(date),
        label: label,
        message: 'Paperwork already exists; no duplicate was created.'
      };
    }

    throw new Error(
      'The weekly session folder already exists but is incomplete. ' +
      'No duplicate paperwork was created.'
    );
  }

  const templates = getPaperworkTemplates_();
  const sessionFolder = weeklyFolder.createFolder(dateCode);

  try {
    const sheetFile = templates.sheet.makeCopy(
      `LBA-Bingo_${dateCode}`,
      sessionFolder
    );

    templates.pdf.makeCopy(
      `LBA-fw2g_${dateCode}.pdf`,
      sessionFolder
    );

    verifyPaperworkSessionFiles_(sessionFolder, dateCode);

    const prefill = prefillNewPaperwork_(
      sheetFile.getId(),
      date,
      label,
      registry
    );

    return {
      ok: true,
      created: true,
      dateString: formatPaperworkIsoDate_(date),
      displayDate: formatPaperworkDisplayDate_(date),
      label: label,
      message: `Created one paperwork set. ${prefill.summary}`
    };
  } catch (error) {
    throw new Error(
      `${dateCode} folder was created, but setup did not complete: ` +
      error.message
    );
  }
}


/**
 * Preserves the standalone application's established prefill behavior.
 */
function prefillNewPaperwork_(newSpreadsheetId, sessionDate, label, registry) {
  const newSpreadsheet = SpreadsheetApp.openById(newSpreadsheetId);
  const coverSheet = newSpreadsheet.getSheetByName('Cover');

  if (!coverSheet) {
    return {
      summary: 'Prefill skipped because the Cover sheet is missing.'
    };
  }

  const completed = [];
  const skipped = [];

  const sessionDateText = [
    padPaperwork2_(sessionDate.getMonth() + 1),
    padPaperwork2_(sessionDate.getDate()),
    padPaperwork2_(sessionDate.getFullYear() % 100)
  ].join('/');

  coverSheet.getRange('C8').setNumberFormat('@');
  coverSheet.getRange('C8').setValue(sessionDateText);
  coverSheet.getRange('C9').setValue(label);
  completed.push('date and session type set');

  const records = getPaperworkSessionRecords_(registry);
  const priorSameType = findMostRecentPriorPaperworkRecord_(
    records,
    sessionDate,
    label
  );

  if (priorSameType) {
    const priorSameTypeSheetId = findArchivedPaperworkSpreadsheetId_(
      priorSameType.date
    );

    if (priorSameTypeSheetId) {
      const nextCoverall = readPriorPaperworkCell_(
        priorSameTypeSheetId,
        'Qtrly Report Data',
        'B34'
      );

      if (nextCoverall.found) {
        coverSheet.getRange('C11').setValue(nextCoverall.value);
        completed.push('Next Coverall Value prefilled');
      } else {
        skipped.push('Next Coverall Value unavailable');
      }
    } else {
      skipped.push('prior same-type paperwork unavailable');
    }
  } else {
    skipped.push('no prior same-type session');
  }

  const priorAnyType = findMostRecentPriorPaperworkRecord_(
    records,
    sessionDate,
    null
  );

  if (priorAnyType) {
    const priorAnySheetId = findArchivedPaperworkSpreadsheetId_(
      priorAnyType.date
    );

    if (priorAnySheetId) {
      const carryovers = readPaperworkCarryoverValues_(priorAnySheetId);

      if (carryovers) {
        coverSheet.getRange('D19').setValue(carryovers.donkey);
        coverSheet.getRange('D20').setValue(carryovers.diamonds);
        coverSheet.getRange('D21').setValue(carryovers.monkey);
        coverSheet.getRange('C30').setValue(carryovers.qoh);
        completed.push('latest prior-session carryovers prefilled');
      } else {
        skipped.push('latest prior-session carryovers unavailable');
      }
    } else {
      skipped.push('latest prior-session paperwork unavailable');
    }
  } else {
    skipped.push('no prior session for carryovers');
  }

  const distinctPriorDates = findDistinctPriorPaperworkDates_(
    records,
    sessionDate
  );

  if (distinctPriorDates.length >= 1) {
    const priorAverage = readPaperworkTrackingValueForDate_(
      distinctPriorDates[0],
      26
    );

    if (priorAverage.found) {
      coverSheet.getRange('D25').setValue(priorAverage.value);
      completed.push('ETabs Prior Average prefilled');
    } else {
      skipped.push(
        priorAverage.reason || 'ETabs Prior Average unavailable'
      );
    }
  } else {
    skipped.push('no prior session for ETabs Prior Average');
  }

  if (distinctPriorDates.length >= 10) {
    const pastNet = readPaperworkTrackingValueForDate_(
      distinctPriorDates[9],
      4
    );

    if (pastNet.found) {
      coverSheet.getRange('D26').setValue(pastNet.value);
      completed.push('ETabs Past Net prefilled');
    } else {
      skipped.push(pastNet.reason || 'ETabs Past Net unavailable');
    }
  } else {
    skipped.push('fewer than ten prior distinct session dates');
  }

  let summary = completed.join(', ') + '.';

  if (skipped.length > 0) {
    summary += ' Skipped: ' + skipped.join('; ') + '.';
  }

  return {
    summary: summary
  };
}


function findMostRecentPriorPaperworkRecord_(
  records,
  sessionDate,
  requiredLabel
) {
  const targetTime = paperworkDateOnlyTime_(sessionDate);

  const matches = records.filter((record) => {
    return (
      record.date &&
      paperworkDateOnlyTime_(record.date) < targetTime &&
      (!requiredLabel || record.label === requiredLabel)
    );
  });

  matches.sort((first, second) => {
    return (
      paperworkDateOnlyTime_(second.date) -
      paperworkDateOnlyTime_(first.date)
    );
  });

  return matches.length ? matches[0] : null;
}


function findDistinctPriorPaperworkDates_(records, sessionDate) {
  const targetTime = paperworkDateOnlyTime_(sessionDate);
  const datesByIso = {};

  records.forEach((record) => {
    if (
      record.date &&
      paperworkDateOnlyTime_(record.date) < targetTime
    ) {
      datesByIso[formatPaperworkIsoDate_(record.date)] = record.date;
    }
  });

  return Object.keys(datesByIso)
    .map((iso) => datesByIso[iso])
    .sort((first, second) => {
      return (
        paperworkDateOnlyTime_(second) -
        paperworkDateOnlyTime_(first)
      );
    });
}

    const quarterFolders = collectPaperworkIterator_(

    const sessionFolders = collectPaperworkIterator_(

    const file = findUniquePaperworkSheet_(

  } catch (error) {

  }
}

function readPaperworkTrackingValueForDate_(sessionDate, valueColumn) {
  try {
    const spreadsheet = SpreadsheetApp.openById(
      PAPERWORK_CONFIG.etabsSpreadsheetId
    );

    const sheet = spreadsheet.getSheetByName(
      getPaperworkTrackingSheetName_(sessionDate)
    );

    if (!sheet || sheet.getLastRow() < 11) {
      return {
        found: false,
        reason: 'ETabs tracking sheet or row unavailable'
      };
    }

    const rowCount = sheet.getLastRow() - 11 + 1;
    const dates = sheet.getRange(11, 1, rowCount, 1).getValues();
    const targetIso = formatPaperworkIsoDate_(sessionDate);

    for (let index = 0; index < dates.length; index += 1) {
      const candidate = coercePaperworkSessionDate_(dates[index][0]);

      if (
        candidate &&
        formatPaperworkIsoDate_(candidate) === targetIso
      ) {
        const rowNumber = 11 + index;
        const readiness = sheet
          .getRange(rowNumber, 2)
          .getDisplayValue()
          .trim();

        if (!readiness) {
          return {
            found: false,
            reason: 'ETabs source row column B is empty'
          };
        }

        return {
          found: true,
          value: sheet.getRange(rowNumber, valueColumn).getValue()
        };
      }
    }

    return {
      found: false,
      reason: 'ETabs date was not found'
    };
  } catch (error) {
    return {
      found: false,
      reason: 'ETabs source was inaccessible'
    };
  }
}


function getPaperworkTrackingSheetName_(date) {

  return `Q${quarter}FY${date.getFullYear()} Tracking`;
}


function archivePastSessionPaperwork_(cutoffDate) {
  const registry = getPaperworkSessionsRegistry_();
  const records = getPaperworkSessionRecords_(registry);
  const results = [];
  const processedKeys = {};

  records.forEach((record) => {
    if (
      !record.date ||
      paperworkDateOnlyTime_(record.date) >=
        paperworkDateOnlyTime_(cutoffDate)
    ) {
      return;
    }

    validatePaperworkLabel_(record.label);

    const key =
      formatPaperworkIsoDate_(record.date) +
      '|' + record.label;

    if (processedKeys[key]) {
      return;
    }

    processedKeys[key] = true;
    results.push(
      archiveOneSession_(
        record.date,
        record.label,
        record.sessionId
      )
    );
  });

  return results;
}


function archiveOneSession_(date, label, sessionId) {
  const dateCode = formatPaperworkMMDDYY_(date);
  const weeklyParent = DriveApp.getFolderById(
    PAPERWORK_CONFIG.weeklyFolders[label]
  );

  const weeklyFolders = collectPaperworkIterator_(
    weeklyParent.getFoldersByName(dateCode)
  );

  if (weeklyFolders.length === 0) {
    return {
      ok: true,
      dateCode: dateCode,
      label: label,
      message: 'Not present in weekly folder; presumed already archived.'
    };
  }

  if (weeklyFolders.length > 1) {
    return {
      ok: false,
      dateCode: dateCode,
      label: label,
      message: 'Multiple weekly session folders were found.'
    };
  }

  const folder = weeklyFolders[0];

  try {
    const inventoryResult = processSessionInventoryBeforeArchive_({
      folder: folder,
      date: date,
      label: label,
      sessionId: sessionId
    });

    if (!inventoryResult.ok) {
      return {
        ok: false,
        dateCode: dateCode,
        label: label,
        message:
          inventoryResult.message +
          ' Paperwork remains in the weekly folder.'
      };
    }

    folder.moveTo(getUniquePaperworkArchiveQuarterFolder_(date));

    const issueCount = Number(inventoryResult.issues || 0);
    return {
      ok: true,
      dateCode: dateCode,
      label: label,
      issues: issueCount,
      message:
        `Inventory processed and archived to ` +
        `${getPaperworkArchiveQuarterName_(date)}.` +
        (issueCount
          ? ` ${issueCount} Inventory Issue(s) require review.`
          : ' No Inventory Issues were created.')
    };
  } catch (error) {
    return {
      ok: false,
      dateCode: dateCode,
      label: label,
      message:
        `${error.message} Paperwork remains in the weekly folder.`
    };
  }
}


function archiveSelectedPaperwork(token, items) {
  requireAdmin_(token);
  validatePaperworkConfiguration_();

  if (!Array.isArray(items) || items.length === 0) {
    throw new Error('No paperwork was selected.');
  }

  const registry = getPaperworkSessionsRegistry_();
  const results = [];

  items.forEach((item) => {
    const date = parsePaperworkMMDDYY_(String(item.folderName || ''));
    const label = normalizeText_(item.label);

    if (!date || !label) {
      results.push({
        ok: false,
        folderName: item.folderName || '(unknown)',
        message: 'Invalid session folder name or session type.'
      });
      return;
    }

    const matches = findPaperworkSessionRecords_(
      registry,
      date,
      label
    );

    if (matches.length !== 1) {
      results.push({
        ok: false,
        folderName: item.folderName,
        message:
          `Expected one Sessions row, but found ${matches.length}.`
      });
      return;
    }

    const result = archiveOneSession_(
      date,
      label,
      matches[0].sessionId
    );

    result.folderName = item.folderName;
    results.push(result);
  });

  return {
    results: results,
    sessions: scanWeeklyPaperworkFolders_()
  };
}


function getPaperworkSessionsRegistry_() {
  const spreadsheet = SpreadsheetApp.openById(
    PAPERWORK_CONFIG.sessionsSpreadsheetId
  );

  const sheet = spreadsheet.getSheetByName(
    PAPERWORK_CONFIG.sessionsSheetName
  );

  if (!sheet) {
    throw new Error('The Sessions worksheet does not exist.');
  }

  const headers = sheet
    .getRange(1, 1, 1, PAPERWORK_REQUIRED_SESSION_HEADERS.length)
    .getDisplayValues()[0];

  PAPERWORK_REQUIRED_SESSION_HEADERS.forEach((expected, index) => {
    if (headers[index] !== expected) {
      throw new Error(
        `Sessions header mismatch in column ${index + 1}. ` +
        `Expected "${expected}".`
      );
    }
  });

  return sheet;
}


function getPaperworkSessionRecords_(sheet) {
  if (sheet.getLastRow() < 2) {
    return [];
  }

  return sheet
    .getRange(
      2,
      1,
      sheet.getLastRow() - 1,
      PAPERWORK_REQUIRED_SESSION_HEADERS.length
    )
    .getValues()
    .map((row, index) => {
      return {
        rowNumber: index + 2,
        sessionId: String(row[0] || ''),
        date: coercePaperworkSessionDate_(row[1]),
        label: String(row[2] || '').trim(),
        active: row[3],
        createdAt: row[4]
      };
    });
}


function findPaperworkSessionRecords_(sheet, date, label) {
  const iso = formatPaperworkIsoDate_(date);

  return getPaperworkSessionRecords_(sheet).filter((record) => {
    return (
      record.date &&
      formatPaperworkIsoDate_(record.date) === iso &&
      record.label === label
    );
  });
}

function getPaperworkTemplates_() {
  const folder = DriveApp.getFolderById(
    PAPERWORK_CONFIG.templateFolderId
  );

  const files = folder.getFiles();
  const sheets = [];
  const pdfs = [];

  while (files.hasNext()) {
    const file = files.next();

    if (
      file.getName().indexOf(PAPERWORK_SHEET_TEMPLATE_PREFIX) === 0 &&
      file.getMimeType() === MimeType.GOOGLE_SHEETS
    ) {
      sheets.push(file);
    }

    if (
      file.getName() === PAPERWORK_PDF_TEMPLATE_NAME &&
      file.getMimeType() === MimeType.PDF
    ) {
      pdfs.push(file);
    }
  }

  if (sheets.length !== 1 || pdfs.length !== 1) {
    throw new Error(
      'The template folder must contain exactly one matching Google Sheet ' +
      'and exactly one LBA-fw2g_TEMPLATE.pdf.'
    );
  }

  return {
    sheet: sheets[0],
    pdf: pdfs[0]
  };
}

function parsePaperworkIsoDate_(text) {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(String(text || ''))) {
    return null;
  }

  const parts = text.split('-').map(Number);
  const date = new Date(parts[0], parts[1] - 1, parts[2]);

  return (
    date.getFullYear() === parts[0] &&
    date.getMonth() === parts[1] - 1 &&
    date.getDate() === parts[2]
  )
    ? date
    : null;
}


function parsePaperworkMMDDYY_(text) {
  if (!PAPERWORK_SESSION_FOLDER_RE.test(String(text || ''))) {
    return null;
  }

  const month = Number(text.slice(0, 2));
  const day = Number(text.slice(2, 4));
  const year = 2000 + Number(text.slice(4, 6));
  const date = new Date(year, month - 1, day);

  return (
    date.getFullYear() === year &&
    date.getMonth() === month - 1 &&
    date.getDate() === day
  )
    ? date
    : null;
}


function coercePaperworkSessionDate_(value) {
  if (value instanceof Date && !isNaN(value.getTime())) {
    return new Date(
      value.getFullYear(),
      value.getMonth(),
      value.getDate()
    );
  }

  const text = String(value || '').trim();
  const iso = parsePaperworkIsoDate_(text);

  if (iso) {
    return iso;
  }

  const match = text.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);

  return match
    ? new Date(Number(match[3]), Number(match[1]) - 1, Number(match[2]))
    : null;
}


function paperworkDateOnlyTime_(date) {

/**
 * SessionInventoryProcessing.js
 * LBA Bingo Inventory Management v1.12
 *
 * Parses completed state paperwork immediately before archival. Processing is
 * attempted once. A Started or Failed marker prevents automatic retry and the
 * paperwork remains in the weekly folder for human recovery.
 */

const SESSION_PROCESSING_SOURCE = 'Session Paperwork Processing';
const SESSION_MARKER_TYPES = [
  'Session Processing Started',
  'Session Processed',
  'Session Processing Failed'
];

const PAPER_ROWS_BY_SESSION = {
  Thursday: [9, 10, 12, 13, 15, 16, 18, 19, 21, 22],
  Friday: [9, 10, 12, 13, 15, 16, 18, 19, 21, 22],
  Sunday: [9, 10, 12, 13, 15, 16, 18, 19, 21, 22, 24, 25, 27, 28]
};

const COMPLETED_PT_RANGES = [
  { sheetName: 'WS-06 PTs (Main)', firstRow: 8, lastRow: 25 },
  { sheetName: 'WS-06 PTs (Flash)', firstRow: 6, lastRow: 17 },
  { sheetName: 'WS-07 Prog PTs', rows: [7, 8, 14, 15, 16, 17, 18, 19, 20, 21, 29, 30] }
];

const CARRYOVER_START_ROWS = [33, 41, 49];


function processSessionInventoryBeforeArchive_(context) {
  const sessionId = normalizeText_(context.sessionId);

  if (!sessionId) {
    return {
      ok: false,
      message: 'No authoritative SessionID could be resolved.'
    };
  }

  const state = getSessionProcessingState_(sessionId);

  if (state === 'Session Processed') {
    return {
      ok: true,
      alreadyProcessed: true,
      message: 'Session inventory was already processed.'
    };
  }

  if (
    state === 'Session Processing Started' ||
    state === 'Session Processing Failed'
  ) {
    return {
      ok: false,
      message:
        `Session inventory has state "${state}". ` +
        'Automatic retry is disabled; manual recovery is required.'
    };
  }

  const dateCode = formatPaperworkMMDDYY_(context.date);
  const file = findUniquePaperworkSheet_(
    context.folder,
    `LBA-Bingo_${dateCode}`
  );

  if (!file) {
    return {
      ok: false,
      message: 'The completed paperwork spreadsheet could not be found.'
    };
  }

  postSessionProcessingMarker_(
    sessionId,
    context.date,
    'Session Processing Started',
    `Started inventory processing from ${file.getName()} (${file.getId()}).`
  );

  try {
    const spreadsheet = SpreadsheetApp.openById(file.getId());

    const paperResult = processSessionPaper_(
      spreadsheet,
      context,
      file
    );

    const ptResult = processSessionPT_(
      spreadsheet,
      context,
      file
    );

    postSessionProcessingMarker_(
      sessionId,
      context.date,
      'Session Processed',
      `Completed inventory processing from ${file.getName()}. ` +
      `${paperResult.transactions} Paper transaction(s); ` +
      `${ptResult.transactions} PT transaction(s); ` +
      `${paperResult.issues + ptResult.issues} issue(s).`
    );

    return {
      ok: true,
      paperTransactions: paperResult.transactions,
      ptTransactions: ptResult.transactions,
      paperIssues: paperResult.issues,
      ptIssues: ptResult.issues,
      issues: paperResult.issues + ptResult.issues,
      message: 'Session inventory processing completed.'
    };
  } catch (error) {
    try {
      postSessionProcessingMarker_(
        sessionId,
        context.date,
        'Session Processing Failed',
        `Inventory processing failed: ${error.message}`
      );
    } catch (markerError) {
      console.error(
        'Unable to record Session Processing Failed marker:',
        markerError
      );
    }

    return {
      ok: false,
      message:
        `Inventory processing failed: ${error.message}`
    };
  }
}


function getSessionProcessingState_(sessionId) {
  let state = '';
  visitTransactionsNewestFirst_(
    ['Inventory_Type', 'Transaction_Type', 'Session_ID', 'Source'],
    (transaction) => {
      if (
        String(transaction.Inventory_Type) === 'System' &&
        String(transaction.Session_ID) === String(sessionId) &&
        String(transaction.Source) === SESSION_PROCESSING_SOURCE &&
        SESSION_MARKER_TYPES.includes(String(transaction.Transaction_Type))
      ) {
        state = String(transaction.Transaction_Type);
        return true;
      }
      return false;
    }
  );
  return state;
}


function postSessionProcessingMarker_(
  sessionId,
  effectiveDate,
  transactionType,
  notes
) {
  return postInventoryTransaction_({
    effectiveDate: effectiveDate,
    inventoryType: 'System',
    productId: '',
    itemId: '',
    formNumber: '',
    serialNumber: '',
    transactionType: transactionType,
    quantityChange: 0,
    unit: 'Session',
    sessionId: sessionId,
    source: SESSION_PROCESSING_SOURCE,
    notes: notes
  });
}


function processSessionPaper_(spreadsheet, context, sourceFile) {
  const sheet = spreadsheet.getSheetByName('WS-01 Paper');
  if (!sheet) throw new Error('Required worksheet WS-01 Paper is missing.');
  const requestedRows = PAPER_ROWS_BY_SESSION[context.label];
  const firstRow = Math.min(...requestedRows);
  const lastRow = Math.max(...requestedRows);
  const values = sheet.getRange(firstRow, 2, lastRow - firstRow + 1, 5).getValues();
  const casesBySerial = new Map(getPaperCases_().filter((row) => normalizeText_(row.Serial_Number)).map((row) => [normalizeText_(row.Serial_Number), row]));
  const entries = [];
  const issues = [];
  const caseUpdates = new Map();

  requestedRows.forEach((rowNumber) => {
    const row = values[rowNumber - firstRow];
    const description = normalizeText_(row[0]);
    const serialNumber = normalizeText_(row[1]);
    const quantity = Number(row[4] || 0);
    if (!serialNumber && quantity === 0) return;
    if (!serialNumber || !Number.isInteger(quantity) || quantity < 0) throw new Error(`Invalid Paper entry on WS-01 row ${rowNumber}.`);
    if (quantity === 0) return;
    const paperCase = casesBySerial.get(serialNumber);
    if (!paperCase) {
      issues.push({inventoryType:'Paper', issueType:'Unknown Serial', productId:'', itemId:'', serialNumber:serialNumber,
        sessionId:context.sessionId, expectedQuantity:0, reportedQuantity:quantity, discrepancyAmount:quantity,
        notes:`WS-01 row ${rowNumber}: ${description}. Source file ${sourceFile.getId()}.`});
      return;
    }
    const previous = caseUpdates.get(String(paperCase.Paper_Case_ID)) || {delta: 0, row: paperCase};
    const balance = Number(paperCase.Current_Packs || 0) + previous.delta;
    if (quantity > balance) issues.push({inventoryType:'Paper', issueType:'Insufficient Inventory', productId:paperCase.Paper_ID,
      itemId:paperCase.Paper_Case_ID, serialNumber:serialNumber, sessionId:context.sessionId, expectedQuantity:balance,
      reportedQuantity:quantity, discrepancyAmount:quantity-balance,
      notes:`Paperwork reports ${quantity} packs used from a current balance of ${balance}. WS-01 row ${rowNumber}.`});
    entries.push({effectiveDate:context.date, inventoryType:'Paper', productId:paperCase.Paper_ID, itemId:paperCase.Paper_Case_ID,
      serialNumber:serialNumber, transactionType:'Sold', quantityChange:-quantity, unit:'Pack', sessionId:context.sessionId,
      source:SESSION_PROCESSING_SOURCE, notes:`${description}; WS-01 row ${rowNumber}; source file ${sourceFile.getId()}.`});
    previous.delta -= quantity;
    caseUpdates.set(String(paperCase.Paper_Case_ID), previous);
  });

  createInventoryIssues_(issues);
  postInventoryTransactions_(entries);
  const statusUpdates = [];
  caseUpdates.forEach(({delta, row}) => {
    const remaining = Number(row.Current_Packs || 0) + delta;
    const updates = {};
    if (remaining <= 0) { updates.Case_Status='Expended'; updates.Expended_Date=context.date; }
    else if (String(row.Case_Status) === 'Unopened') { updates.Case_Status='Opened'; updates.Opened_Date=context.date; }
    if (Object.keys(updates).length) statusUpdates.push({rowNumber:row._rowNumber, updates:updates});
  });
  updateRecordRows_('Paper_Cases', statusUpdates);
  return {
    transactions: entries.length,
    issues: issues.length
  };
}

function processSessionPT_(spreadsheet, context, sourceFile) {
  const observations = [];

  COMPLETED_PT_RANGES.forEach((definition) => {
    const sheet = spreadsheet.getSheetByName(definition.sheetName);
    if (!sheet) {
      throw new Error(`Required worksheet ${definition.sheetName} is missing.`);
    }

    const rows = definition.rows ||
      createIntegerRange_(definition.firstRow, definition.lastRow);
    const firstRow = Math.min(...rows);
    const lastRow = Math.max(...rows);
    const values = sheet
      .getRange(firstRow, 4, lastRow - firstRow + 1, 2)
      .getDisplayValues();

    rows.forEach((rowNumber) => {
      const row = values[rowNumber - firstRow];
      const serialNumber = normalizeText_(row[0]);
      const formNumber = normalizeFormNumber_(row[1]);
      if (
        !serialNumber ||
        !formNumber ||
        /^#N\/A$/i.test(serialNumber)
      ) {
        return;
      }

      observations.push({
        kind: 'completed',
        formNumber: formNumber,
        serialNumber: serialNumber,
        location: `${definition.sheetName} row ${rowNumber}`
      });
    });
  });

  const main = spreadsheet.getSheetByName('WS-06 PTs (Main)');
  if (!main) {
    throw new Error('Required worksheet WS-06 PTs (Main) is missing.');
  }

  const firstCarryoverRow = Math.min(...CARRYOVER_START_ROWS);
  const lastCarryoverRow = Math.max(...CARRYOVER_START_ROWS) + 4;
  const values = main
    .getRange(
      firstCarryoverRow,
      4,
      lastCarryoverRow - firstCarryoverRow + 1,
      3
    )
    .getValues();

  CARRYOVER_START_ROWS.forEach((startRow) => {
    const start = values[startRow - firstCarryoverRow];
    const previousSales = values[startRow + 3 - firstCarryoverRow];
    const currentSales = values[startRow + 4 - firstCarryoverRow];
    const serialNumber = normalizeText_(start[0]);
    const formNumber = normalizeFormNumber_(start[1]);
    if (!serialNumber || !formNumber) return;

    observations.push({
      kind: 'carryover',
      formNumber: formNumber,
      serialNumber: serialNumber,
      ticketCount: Number(start[2] || 0),
      totalSales:
        Number(previousSales[1] || 0) +
        Number(currentSales[1] || 0),
      location: `Carry-over block row ${startRow}`
    });
  });

  observations.forEach((observation) => {
    const resolved = resolvePTGameByForm_(
      observation.formNumber,
      { includeInactive: true }
    );
    observation.gameId = resolved ? resolved.gameId : '';
    observation.primaryFormNumber = resolved
      ? resolved.primaryFormNumber
      : observation.formNumber;
  });

  const dealsByKey = new Map();
  const anonymousDealsByGameId = new Map();

  getPTDeals_().forEach((deal) => {
    const gameId = String(deal.PT_Game_ID);
    const serialNumber = normalizeText_(deal.Serial_Number);

    if (serialNumber) {
      dealsByKey.set(`${gameId}::${serialNumber}`, deal);
      return;
    }

    if (String(deal.Deal_Status) !== 'Unbundled') return;

    const anonymousDeals = anonymousDealsByGameId.get(gameId) || [];
    anonymousDeals.push(deal);
    anonymousDealsByGameId.set(gameId, anonymousDeals);
  });

  const plannedStatuses = new Map();
  const dealUpdates = [];
  const transactions = [];
  const issues = [];
  const balanceDeltas = new Map();

  observations.forEach((observation) => {
    if (
      observation.kind === 'carryover' &&
      observation.totalSales > observation.ticketCount
    ) {
      issues.push({
        inventoryType: 'PT',
        issueType: 'Unexpected Deal',
        productId: observation.gameId,
        itemId: '',
        formNumber: observation.primaryFormNumber,
        serialNumber: observation.serialNumber,
        sessionId: context.sessionId,
        expectedQuantity: 1,
        reportedQuantity: 1,
        discrepancyAmount: 0,
        notes:
          `Carry-over sales ${observation.totalSales} exceed ` +
          `ticket count ${observation.ticketCount}.`
      });
      return;
    }

    const dealKey = observation.gameId
      ? `${observation.gameId}::${observation.serialNumber}`
      : '';
    let deal = dealKey ? dealsByKey.get(dealKey) : null;
    let claimedAnonymousDeal = false;

    if (!deal && observation.gameId) {
      const anonymousDeals =
        anonymousDealsByGameId.get(String(observation.gameId)) || [];
      deal = anonymousDeals.shift() || null;

      if (deal) {
        claimedAnonymousDeal = true;
        dealsByKey.set(dealKey, deal);
      }
    }

    if (!deal) {
      issues.push({
        inventoryType: 'PT',
        issueType: 'Unknown Deal',
        productId: observation.gameId,
        itemId: '',
        formNumber: observation.primaryFormNumber,
        serialNumber: observation.serialNumber,
        sessionId: context.sessionId,
        expectedQuantity: 0,
        reportedQuantity: 1,
        discrepancyAmount: 1,
        notes:
          `${observation.location}; source file ${sourceFile.getId()}.`
      });
      return;
    }

    const dealId = String(deal.PT_Deal_ID);
    const priorStatus = plannedStatuses.get(dealId) ||
      String(deal.Deal_Status);
    let targetStatus;
    let transactionType;
    let quantityChange;
    let notes;

    if (observation.kind === 'completed') {
      if (priorStatus === 'Sold') {
        issues.push({
          inventoryType: 'PT',
          issueType: 'Duplicate Sale',
          productId: deal.PT_Game_ID,
          itemId: dealId,
          formNumber: observation.primaryFormNumber,
          serialNumber: observation.serialNumber,
          sessionId: context.sessionId,
          expectedQuantity: 1,
          reportedQuantity: 1,
          discrepancyAmount: 0,
          notes: observation.location + '.'
        });
        return;
      }

      if (!['Unbundled', 'Bundled', 'Open'].includes(priorStatus)) {
        issues.push({
          inventoryType: 'PT',
          issueType: 'Unavailable Deal',
          productId: deal.PT_Game_ID,
          itemId: dealId,
          formNumber: observation.primaryFormNumber,
          serialNumber: observation.serialNumber,
          sessionId: context.sessionId,
          expectedQuantity: 1,
          reportedQuantity: 1,
          discrepancyAmount: 0,
          notes:
            `Deal status was ${priorStatus}; ${observation.location}.`
        });
        return;
      }

      targetStatus = 'Sold';
      transactionType = 'Sold';
      quantityChange = priorStatus === 'Open' ? 0 : -1;
      notes =
        `${observation.location}; source file ${sourceFile.getId()}.`;
    } else {
      targetStatus = 'Open';
      if (priorStatus === targetStatus) return;

      if (!['Unbundled', 'Bundled'].includes(priorStatus)) {
        issues.push({
          inventoryType: 'PT',
          issueType: 'Unavailable Deal',
          productId: deal.PT_Game_ID,
          itemId: dealId,
          formNumber: observation.primaryFormNumber,
          serialNumber: observation.serialNumber,
          sessionId: context.sessionId,
          expectedQuantity: 1,
          reportedQuantity: 1,
          discrepancyAmount: 0,
          notes: `Carry-over deal status was ${priorStatus}.`
        });
        return;
      }

      transactionType = 'Opened';
      quantityChange = -1;
      notes =
        `${observation.location}; ` +
        `${observation.totalSales}/${observation.ticketCount} sold; ` +
        `source file ${sourceFile.getId()}.`;
    }

    const updates = {
      Form_Number: observation.primaryFormNumber,
      Deal_Status: targetStatus
    };

    if (claimedAnonymousDeal) {
      updates.Serial_Number = observation.serialNumber;
    }

    if (targetStatus === 'Sold') {
      updates.Expended_Date = context.date;
      updates.Expended_Session_ID = context.sessionId;
    }

    dealUpdates.push({
      rowNumber: deal._rowNumber,
      updates: updates
    });
    recordPTStatusChange_(
      balanceDeltas,
      deal.PT_Game_ID,
      priorStatus,
      targetStatus
    );
    plannedStatuses.set(dealId, targetStatus);

    transactions.push({
      effectiveDate: context.date,
      inventoryType: 'PT',
      productId: deal.PT_Game_ID,
      itemId: dealId,
      formNumber: observation.primaryFormNumber,
      serialNumber: observation.serialNumber,
      transactionType: transactionType,
      quantityChange: quantityChange,
      unit: 'Deal',
      sessionId: context.sessionId,
      source: SESSION_PROCESSING_SOURCE,
      notes: notes
    });
  });

  createInventoryIssues_(issues);
  updateRecordRows_('PT_Deals', dealUpdates);
  applyPTCatalogBalanceDeltas_(balanceDeltas);
  postInventoryTransactions_(transactions);

  return {
    transactions: transactions.length,
    issues: issues.length
  };
}


function createIntegerRange_(first, last) {
  const output = [];

  for (let value = first; value <= last; value += 1) {
    output.push(value);
  }

  return output;
}

/**
 * Creates the complete v1 inventory workbook schema in the bound spreadsheet.
 * Safe initialization: aborts rather than overwriting any existing schema sheet.
 */
function initializeInventoryWorkbook() {
  assertAdvancedSheetsService_();

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error('No active spreadsheet. Run this function from the bound inventory workbook.');

  const schemaNames = Object.keys(INVENTORY_SCHEMA_V1.sheets);
  const existingSchemaSheets = schemaNames.filter(name => ss.getSheetByName(name));
  if (existingSchemaSheets.length) {
    throw new Error(
      'Initialization stopped. These schema sheets already exist: ' + existingSchemaSheets.join(', ') +
      '. Do not rerun initialization against an existing inventory workbook.'
    );
  }

  const existingSheets = ss.getSheets();
  const reusableBlankSheet = existingSheets.length === 1 && isSheetBlank_(existingSheets[0])
    ? existingSheets[0]
    : null;

  const created = [];
  try {
    schemaNames.forEach((name, index) => {
      let sheet;
      if (index === 0 && reusableBlankSheet) {
        sheet = reusableBlankSheet;
        sheet.setName(name);
      } else {
        sheet = ss.insertSheet(name);
      }
      created.push(sheet);
      initializeOneSheet_(sheet, name, INVENTORY_SCHEMA_V1.sheets[name]);
    });

    ss.setActiveSheet(ss.getSheetByName('Dashboard'));
    SpreadsheetApp.flush();

    createNativeTables_();
    applyCalculatedColumns_();
    SpreadsheetApp.flush();

    const report = validateInventorySchemaV1(false);
    if (!report.valid) {
      throw new Error('Initialization completed but validation failed:\n' + report.errors.join('\n'));
    }

    tryShowSchemaDialog_('LBA Bingo Inventory', 'Schema v1 was initialized and validated successfully.');
    return report;
  } catch (error) {
    throw new Error('Inventory schema initialization failed: ' + error.message);
  }
}

/**
 * Validates the workbook against schema v1 without changing data.
 * @param {boolean=} showDialog Whether to show a summary dialog. Defaults to true.
 * @return {{valid:boolean, version:string, errors:string[], warnings:string[]}}
 */
/**
 * Repairs v1 validation/format rules on an already initialized workbook.
 * This is safe for the validation failure seen after native Table creation;
 * it does not recreate sheets or alter inventory records.
 */
function repairInventorySchemaV1Validation() {
  rebuildNativeTablesWithTypedColumns_();
  restoreMissingV1Settings_();
  applyCalculatedColumns_();
  SpreadsheetApp.flush();
  return validateInventorySchemaV1(false);
}

function validateInventorySchemaV1(showDialog = true) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const errors = [];
  const warnings = [];

  Object.entries(INVENTORY_SCHEMA_V1.sheets).forEach(([sheetName, definition]) => {
    const sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      errors.push(`Missing sheet: ${sheetName}`);
      return;
    }

    if (definition.headers.length) {
      const actual = sheet.getRange(1, 1, 1, definition.headers.length).getDisplayValues()[0];
      definition.headers.forEach((expected, i) => {
        if (actual[i] !== expected) {
          errors.push(`${sheetName}: expected header ${expected} in column ${i + 1}; found ${actual[i] || '(blank)'}.`);
        }
      });

      const extraHeader = sheet.getRange(1, definition.headers.length + 1).getDisplayValue();
      if (extraHeader) warnings.push(`${sheetName}: unexpected extra header after the defined v1 columns: ${extraHeader}.`);

      validateConfiguredColumns_(sheet, sheetName, definition, errors, warnings);
    }
  });

  validateSettings_(ss, errors, warnings);
  validateNativeTables_(ss, errors, warnings);

  const report = {
    valid: errors.length === 0,
    version: INVENTORY_SCHEMA_V1.version,
    errors,
    warnings,
  };

  console.log(JSON.stringify(report, null, 2));

  if (showDialog) {
    const title = report.valid ? 'Schema v1 validation passed' : 'Schema v1 validation failed';
    const body = [
      report.valid ? 'No schema errors were found.' : `${errors.length} error(s) were found.`,
      warnings.length ? `${warnings.length} warning(s) were found.` : 'No warnings were found.',
      '',
      errors.slice(0, 12).join('\n'),
      errors.length > 12 ? `\n...and ${errors.length - 12} more error(s). See the execution log.` : '',
      warnings.length ? '\nWarnings:\n' + warnings.slice(0, 8).join('\n') : '',
    ].join('\n');
    tryShowSchemaDialog_(title, body);
  }

  return report;
}

function initializeOneSheet_(sheet, sheetName, definition) {
  sheet.clear();
  sheet.clearConditionalFormatRules();

  if (sheetName === 'Dashboard') {
    sheet.getRange('A1').setValue('LBA Bingo Inventory Dashboard')
      .setFontSize(18).setFontWeight('bold').setFontColor('#ffffff').setBackground('#1f4e78');
    sheet.getRange('A3').setValue('Dashboard formulas and layout will be added after catalog and opening inventory data are established.')
      .setFontStyle('italic').setFontColor('#666666');
    sheet.setColumnWidth(1, 820);
    sheet.setFrozenRows(1);
    sheet.setTabColor('#1f4e78');
    return;
  }

  const headers = definition.headers;
  ensureSheetSize_(sheet, INVENTORY_SCHEMA_V1.tableRows, Math.max(headers.length, 3));
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  invalidateInventoryHeaderCache_(sheet);
  sheet.getRange(1, 1, 1, headers.length)
    .setFontWeight('bold')
    .setFontColor('#ffffff')
    .setBackground('#1f4e78')
    .setHorizontalAlignment('center')
    .setWrap(true);
  sheet.setFrozenRows(1);
  sheet.setTabColor(sheetName.includes('Catalog') ? '#6aa84f' : sheetName === 'Settings' ? '#999999' : '#3d85c6');

  definition.widths.forEach((width, i) => sheet.setColumnWidth(i + 1, width));
  sheet.getRange(2, 1, INVENTORY_SCHEMA_V1.tableRows - 1, headers.length).setVerticalAlignment('middle');

  applyDefinitionValidation_(sheet, definition);
  applyDefinitionFormats_(sheet, definition);

  if (definition.initialRows && definition.initialRows.length) {
    sheet.getRange(2, 1, definition.initialRows.length, definition.headers.length).setValues(definition.initialRows);
  }
}

function applyDefinitionValidation_(sheet, definition) {
  const totalDataRows = INVENTORY_SCHEMA_V1.tableRows - 1;

  Object.entries(definition.dropdowns || {}).forEach(([header, options]) => {
    const col = columnIndex_(definition.headers, header);
    if (!col) return;
    const rule = SpreadsheetApp.newDataValidation()
      .requireValueInList(options, true)
      .setAllowInvalid(false)
      .setHelpText(`Choose one of: ${options.join(', ')}`)
      .build();
    sheet.getRange(2, col, totalDataRows, 1).setDataValidation(rule);
  });

  (definition.checkboxes || []).forEach(header => {
    const col = columnIndex_(definition.headers, header);
    if (col) sheet.getRange(2, col, totalDataRows, 1).insertCheckboxes();
  });

  (definition.integers || []).forEach(header => {
    const col = columnIndex_(definition.headers, header);
    if (!col) return;
    const range = sheet.getRange(2, col, totalDataRows, 1);
    const rule = SpreadsheetApp.newDataValidation()
      .requireFormulaSatisfied(`=OR(${range.getCell(1, 1).getA1Notation()}="",AND(ISNUMBER(${range.getCell(1, 1).getA1Notation()}),MOD(${range.getCell(1, 1).getA1Notation()},1)=0))`)
      .setAllowInvalid(false)
      .setHelpText('Enter a whole number. Transaction quantities must always be integers.')
      .build();
    range.setDataValidation(rule);
  });
}

function applyDefinitionFormats_(sheet, definition) {
  const rows = INVENTORY_SCHEMA_V1.tableRows - 1;
  formatColumns_(sheet, definition, definition.integers || [], '0', rows);
  formatColumns_(sheet, definition, definition.decimals || [], '0.0###', rows);
  formatColumns_(sheet, definition, definition.currency || [], '$#,##0.00', rows);
  formatColumns_(sheet, definition, definition.percent || [], '0.00%', rows);
  formatColumns_(sheet, definition, definition.dates || [], 'm/d/yyyy', rows);
  formatColumns_(sheet, definition, definition.timestamps || [], 'm/d/yyyy h:mm:ss am/pm', rows);
}

function formatColumns_(sheet, definition, headers, format, rows) {
  headers.forEach(header => {
    const col = columnIndex_(definition.headers, header);
    if (col) sheet.getRange(2, col, rows, 1).setNumberFormat(format);
  });
}

function applyCalculatedColumns_() {

    });
  });

function addNativeTablesWithTypedColumns_(replaceExisting) {
  assertAdvancedSheetsService_();
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  if (replaceExisting) {
    const current = Sheets.Spreadsheets.get(ss.getId(), {includeGridData: false});
    const deleteRequests = [];
    (current.sheets || []).forEach(apiSheet => {
      (apiSheet.tables || []).forEach(table => {
        deleteRequests.push({deleteTable: {tableId: table.tableId}});
      });
    });
    if (deleteRequests.length) {
      Sheets.Spreadsheets.batchUpdate({requests: deleteRequests}, ss.getId());
    }
  }

  const requests = [];
  Object.entries(INVENTORY_SCHEMA_V1.sheets).forEach(([sheetName, definition]) => {
    if (!definition.tableName || !definition.headers.length) return;
    const sheet = ss.getSheetByName(sheetName);
    const rowCount = Math.max(definition.initialRows ? definition.initialRows.length + 2 : 20, 20);
    const columnProperties = buildTableColumnProperties_(definition);
    requests.push({
      addTable: {
        table: {
          name: definition.tableName,
          range: {
            sheetId: sheet.getSheetId(),
            startRowIndex: 0,
            endRowIndex: rowCount,
            startColumnIndex: 0,
            endColumnIndex: definition.headers.length,
          },
          columnProperties: columnProperties,
        },
      },
    });
  });
  if (requests.length) Sheets.Spreadsheets.batchUpdate({requests}, ss.getId());
}

function buildTableColumnProperties_(definition) {
  return definition.headers.map((header, zeroBasedIndex) => {
    const property = {
      columnIndex: zeroBasedIndex,
      columnName: header,
    };
    if ((definition.dropdowns || {})[header]) {
      property.columnType = 'DROPDOWN';
      property.dataValidationRule = {
        condition: {
          type: 'ONE_OF_LIST',
          values: definition.dropdowns[header].map(value => ({userEnteredValue: value})),
        },
      };
    } else if ((definition.checkboxes || []).includes(header)) {
      // The Sheets REST API does not currently accept CHECKBOX as a
      // TableColumnProperties ColumnType. Leave the native table column
      // untyped, then ordinary checkbox validation is applied only to
      // cells OUTSIDE the native table range by Apps Script. For Active
      // catalog fields, TRUE/FALSE values remain valid in the table itself.
    }
    return property;
  });
}

function validateConfiguredColumns_(sheet, sheetName, definition, errors, warnings) {
  Object.keys(definition.formulas || {}).forEach(header => {
    const col = columnIndex_(definition.headers, header);
    if (!sheet.getRange(2, col).getFormula()) {
      errors.push(`${sheetName}.${header}: missing calculated-column formula in row 2.`);
    }
  });
}

function restoreMissingV1Settings_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Settings');
  if (!sheet) throw new Error('Missing required sheet: Settings');

  const lastRow = sheet.getLastRow();
  const existingValues = lastRow > 1
    ? sheet.getRange(2, 1, lastRow - 1, 3).getValues()
    : [];
  const existing = new Set(existingValues.filter(row => row[0]).map(row => String(row[0])));
  const missing = INVENTORY_SCHEMA_V1.sheets.Settings.initialRows.filter(row => !existing.has(row[0]));

  if (missing.length) {
    const startRow = Math.max(sheet.getLastRow() + 1, 2);
    sheet.getRange(startRow, 1, missing.length, 3).setValues(missing);
  }
}

function validateSettings_(ss, errors, warnings) {
  const sheet = ss.getSheetByName('Settings');
  if (!sheet) return;

  const dataRowCount = Math.max(0, sheet.getLastRow() - 1);
  const values = dataRowCount > 0
    ? sheet.getRange(2, 1, dataRowCount, 2).getDisplayValues()
    : [];
  const settings = Object.fromEntries(values.filter(row => row[0]).map(row => [row[0], row[1]]));

  const required = INVENTORY_SCHEMA_V1.sheets.Settings.initialRows.map(row => row[0]);
  required.forEach(key => {
    if (!(key in settings)) errors.push(`Settings: missing required setting ${key}.`);
  });

  if (settings.Schema_Version !== INVENTORY_SCHEMA_V1.version) {
    errors.push(`Settings.Schema_Version: expected ${INVENTORY_SCHEMA_V1.version}; found ${settings.Schema_Version || '(blank)'}.`);
  }

  ['Inventory_Start_Date', 'Thursday_Paperwork_Folder_ID', 'Friday_Paperwork_Folder_ID', 'Sunday_Paperwork_Folder_ID', 'Paperwork_Template_Folder_ID']
    .forEach(key => {
      if (key in settings && !settings[key]) warnings.push(`Settings.${key} is currently blank.`);
    });
}

function validateNativeTables_(ss, errors, warnings) {
  try {
    assertAdvancedSheetsService_();
    const response = Sheets.Spreadsheets.get(ss.getId(), {includeGridData: false});
    const found = new Set();
    (response.sheets || []).forEach(apiSheet => {
      (apiSheet.tables || []).forEach(table => found.add(table.name));
    });

    Object.values(INVENTORY_SCHEMA_V1.sheets).forEach(definition => {
      if (definition.tableName && !found.has(definition.tableName)) {
        errors.push(`Missing native Google Sheets table: ${definition.tableName}.`);
      }
    });
  } catch (error) {
    warnings.push('Native table validation could not run: ' + error.message);
  }
}

function tryShowSchemaDialog_(title, body) {
  try {
    const ui = SpreadsheetApp.getUi();
    ui.alert(title, body, ui.ButtonSet.OK);
  } catch (error) {
    console.log(`${title}: ${body}`);
  }
}

function assertAdvancedSheetsService_() {
  if (typeof Sheets === 'undefined' || !Sheets.Spreadsheets) {
    throw new Error(
      'The Advanced Google Sheets service is not enabled. In Apps Script, open Services, click +, add Google Sheets API, and rerun.'
    );
  }
}

function columnIndex_(headers, header) {
  const index = headers.indexOf(header);
  return index < 0 ? 0 : index + 1;
}

function ensureSheetSize_(sheet, rows, columns) {
  if (sheet.getMaxRows() < rows) sheet.insertRowsAfter(sheet.getMaxRows(), rows - sheet.getMaxRows());
  if (sheet.getMaxColumns() < columns) sheet.insertColumnsAfter(sheet.getMaxColumns(), columns - sheet.getMaxColumns());
}

function isSheetBlank_(sheet) {
  return sheet.getLastRow() === 1 && sheet.getLastColumn() === 1 && sheet.getRange('A1').isBlank();
}

/**
 * SchemaPatchV114.js
 * One-time v1.14 schema migration.
 * Scope: Adds Inventory_Issues resolution-history columns and extends the native Table.
 * Current as of v1.20.
 */
function applySchemaV114Patch() {
  const sheet = getSheetOrThrow_('Inventory_Issues');
  const currentHeaders = sheet
    .getRange(1, 1, 1, sheet.getLastColumn())
    .getDisplayValues()[0];

  const requiredHeaders = [
    'Resolved_At',
    'Resolution_Notes'
  ];

  const missingHeaders = requiredHeaders.filter((header) => {
    return !currentHeaders.includes(header);
  });

  if (!missingHeaders.length) {
    return {
      ok: true,
      changed: false,
      message: 'V1.14 Inventory_Issues columns already exist.'
    };
  }

  const firstNewColumn = currentHeaders.length + 1;
  const requiredColumnCount = currentHeaders.length + missingHeaders.length;

  if (sheet.getMaxColumns() < requiredColumnCount) {
    sheet.insertColumnsAfter(
      sheet.getMaxColumns(),
      requiredColumnCount - sheet.getMaxColumns()
    );
  }

  sheet
    .getRange(1, firstNewColumn, 1, missingHeaders.length)
    .setValues([missingHeaders])
    .setFontWeight('bold')
    .setFontColor('#ffffff')
    .setBackground('#1f4e78')
    .setHorizontalAlignment('center')
    .setWrap(true);

  missingHeaders.forEach((header, offset) => {
    const column = firstNewColumn + offset;

    sheet.setColumnWidth(
      column,
      header === 'Resolved_At' ? 155 : 300
    );

    if (header === 'Resolved_At') {
      sheet
        .getRange(2, column, sheet.getMaxRows() - 1, 1)
        .setNumberFormat('m/d/yyyy h:mm:ss am/pm');
    }
  });

  const table = getNativeTableForSheet_(sheet);

  if (!table) {
    throw new Error(
      'Inventory_Issues native Table was not found. ' +
      'The worksheet columns were added, but the Table must be repaired.'
    );
  }

  const updatedTable = JSON.parse(JSON.stringify(table));
  updatedTable.range.endColumnIndex = requiredColumnCount;
  updatedTable.columnProperties = [
    ...(updatedTable.columnProperties || []),
    ...missingHeaders.map((header, offset) => {
      return {
        columnIndex: firstNewColumn - 1 + offset,
        columnName: header
      };
    })
  ];

  Sheets.Spreadsheets.batchUpdate(
    {
      requests: [
        {
          updateTable: {
            table: updatedTable,
            fields: 'range,columnProperties'
          }
        }
      ]
    },
    getInventorySpreadsheet_().getId()
  );

  SpreadsheetApp.flush();

  return {
    ok: true,
    changed: true,
    addedHeaders: missingHeaders,
    message: 'V1.14 Inventory_Issues columns were added.'
  };
}
